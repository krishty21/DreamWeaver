    25→function clamp(n: unknown, lo = 0, hi = 1): number {
    26→  const v = typeof n === "number" ? n : Number(n);
    27→  if (!isFinite(v)) return lo;
    28→  return Math.max(lo, Math.min(hi, v));
    29→}

    31→// Normalize common model quirks before parsing: smart quotes, // and /* */
    32→// comments, BOM, zero-width chars.
    33→function normalizeJSONSource(t: string): string {
    34→  return t
    35→    .replace(/\uFEFF/g, "")
    36→    .replace(/[\u200B-\u200D\u2060]/g, "")
    37→    .replace(/[\u201C\u201D]/g, '"')
    38→    .replace(/[\u2018\u2019]/g, "'")
    39→    .replace(/[\u2013\u2014]/g, "-")
    40→    // line comments (only outside strings — the crude strip below is safe
    41→    // enough for model output where // almost always starts a comment)
    42→    .replace(/^\s*\/\/.*$/gm, "")
    43→    .replace(/\/\*[\s\S]*?\*\//g, "");
    44→}

    46→// Attempt JSON.parse with progressively more aggressive repairs.
    47→function parseWithRepairs(s: string): any | null {
    48→  // 1. direct parse
    49→  try {
    50→    return JSON.parse(s);
    51→  } catch {}
    52→
    53→  // 2. trailing commas
    54→  try {
    55→    return JSON.parse(s.replace(/,\s*([}\]])/g, "$1"));
    56→  } catch {}
    57→
    58→  // 3. truncated output — cut back to the last value that parsed cleanly,
    59→  //    then close any open brackets/braces.
    60→  try {
    61→    const repaired = repairTruncated(s.replace(/,\s*([}\]])/g, "$1"));
    62→    if (repaired) return JSON.parse(repaired);
    63→  } catch {}
    64→
    65→  return null;
    66→}

    68→// Repair a truncated JSON document: terminate a dangling string, strip the
    69→// tail after the last complete value if possible, then close open structures.
    70→function repairTruncated(s: string): string | null {
    71→  // Walk the string tracking string-state and open brackets.
    72→  let inString = false;
    73→  let escape = false;
    74→  const stack: string[] = [];
    75→  let lastSafe = -1; // index of the last "clean value boundary" outside strings
    76→  for (let i = 0; i < s.length; i++) {
    77→    const ch = s[i];
    78→    if (escape) {
    79→      escape = false;
    80→      continue;
    81→    }
    82→    if (ch === "\\" && inString) {
    83→      escape = true;
    84→      continue;
    85→    }
    86→    if (ch === '"') {
    87→      inString = !inString;
    88→      continue;
    89→    }
    90→    if (inString) continue;
    91→    if (ch === "{" || ch === "[") stack.push(ch);
    92→    else if (ch === "}" || ch === "]") stack.pop();
    93→    else if (ch === "," && stack.length > 0) lastSafe = i;
    94→  }
    95→  if (stack.length === 0) return null; // nothing to close — not truncation-shaped
    96→
    97→  // Cut at the last comma that sat inside an open structure (drops the
    98→  // incomplete trailing value), unless that would discard the only content.
    99→  let out: string;
   100→  if (lastSafe > 0) {
   101→    out = s.slice(0, lastSafe).replace(/[,\s]+$/, "");
   102→  } else {
   103→    // No safe comma boundary — keep everything; if we're mid-string,
   104→    // terminate the string so the document can still be closed.
   105→    out = s.replace(/[,\s]+$/, "");
   106→    if (inString) out += '"';
   107→  }
   108→
   109→  // Recompute the open-structure stack on the trimmed source.
   110→  let inStr2 = false;
   111→  let esc2 = false;
   112→  const stack2: string[] = [];
   113→  for (let i = 0; i < out.length; i++) {
   114→    const ch = out[i];
   115→    if (esc2) {
   116→      esc2 = false;
   117→      continue;
   118→    }
   119→    if (ch === "\\" && inStr2) {
   120→      esc2 = true;
   121→      continue;
   122→    }
   123→    if (ch === '"') {
   124→      inStr2 = !inStr2;
   125→      continue;
   126→    }
   127→    if (inStr2) continue;
   128→    if (ch === "{" || ch === "[") stack2.push(ch);
   129→    else if (ch === "}" || ch === "]") stack2.pop();
   130→  }
   131→  if (inStr2) return null; // still malformed (e.g. stray quote) — give up
   132→  while (stack2.length > 0) {
   133→    const open = stack2.pop();
   134→    out += open === "{" ? "}" : "]";
   135→  }
   136→  return out;
   137→}

   139→// Try to extract a JSON object from a model response that may contain
   140→// surrounding prose / code fences / truncation.
   141→function extractJSON(text: string): any | null {
   142→  if (!text) return null;
   143→  // strip code fences
   144→  let t = text.trim();
   145→  const fence = t.match(/```(?:json)?\s*([\s\S]*?)```/i);
   146→  if (fence) t = fence[1].trim();
   147→  t = normalizeJSONSource(t);
   148→
   149→  const firstBrace = t.indexOf("{");
   150→  const firstBracket = t.indexOf("[");
   151→  // A top-level array starts before any object — try the array first.
   152→  const arrayFirst =
   153→    firstBracket !== -1 && (firstBrace === -1 || firstBracket < firstBrace);
   154→  if (arrayFirst) {
   155→    const lastBracket = t.lastIndexOf("]");
   156→    const candidate =
   157→      lastBracket > firstBracket
   158→        ? t.slice(firstBracket, lastBracket + 1) // possibly complete array
   159→        : t.slice(firstBracket); // truncated array (final ] dropped)
   160→    const parsedArr = parseWithRepairs(candidate);
   161→    if (parsedArr !== null) return parsedArr;
   162→  }
   163→
   164→  // Object extraction: first { ... last }, with truncation fallbacks.
   165→  const first = firstBrace;
   166→  const last = t.lastIndexOf("}");
   167→  if (first === -1) return null;
   168→  if (last > first) {
   169→    const parsed = parseWithRepairs(t.slice(first, last + 1));
   170→    if (parsed !== null) return parsed;
   171→  }
   172→  // Last resort: everything after the first { is a truncated object.
   173→  return parseWithRepairs(t.slice(first));
   174→}

   479→function clampDelta(d: any): ProposedDelta {
   480→  const out: ProposedDelta = {};
   481→  if (typeof d.fear === "number") out.fear = clamp(d.fear, 0, 100);
   482→  if (typeof d.lucidity === "number") out.lucidity = clamp(d.lucidity, 0, 100);
   483→  if (typeof d.stability === "number") out.stability = clamp(d.stability, 0, 100);
   484→  if (typeof d.agency === "number") out.agency = clamp(d.agency, 0, 100);
   485→  if (Array.isArray(d.discoveredMotifs))
   486→    out.discoveredMotifs = d.discoveredMotifs.map((s: any) => String(s).slice(0, 60).toLowerCase()).slice(0, 8);
   487→  if (typeof d.visitedScene === "string") out.visitedScene = String(d.visitedScene).slice(0, 80);
   488→  if (Array.isArray(d.inventoryAdd))
   489→    out.inventoryAdd = d.inventoryAdd.map((s: any) => String(s).slice(0, 60)).slice(0, 5);
   490→  if (
   491→    d.phase === "opening" ||
   492→    d.phase === "developing" ||
   493→    d.phase === "climax" ||
   494→    d.phase === "resolving"
   495→  )
   496→    out.phase = d.phase;
   497→  if (
   498→    d.ending === "collapse" ||
   499→    d.ending === "escape" ||
   500→    d.ending === "control" ||
   501→    d.ending === "unresolved" ||
   502→    d.ending === "transformed"
   503→  )
   504→    out.ending = d.ending;
   505→  if (typeof d.reasoning === "string") out.reasoning = String(d.reasoning).slice(0, 400);
   506→  return out;
   507→}

   537→// Extract the partial sceneText value from a partial JSON buffer.
   538→// Returns { text, complete } where complete=true if the closing quote was seen.
   539→function extractPartialSceneText(buffer: string): { text: string; complete: boolean } {
   540→  const key = `"sceneText"`;
   541→  const idx = buffer.indexOf(key);
   542→  if (idx === -1) return { text: "", complete: false };
   543→
   544→  // skip whitespace
   545→  let i = idx + key.length;
   546→  while (i < buffer.length && /\s/.test(buffer[i])) i++;
   547→  if (i >= buffer.length || buffer[i] !== ":") return { text: "", complete: false };
   548→  i++;
   549→  while (i < buffer.length && /\s/.test(buffer[i])) i++;
   550→  if (i >= buffer.length) return { text: "", complete: false };
   551→  if (buffer[i] !== '"') return { text: "", complete: false };
   552→
   553→  // string content
   554→  i++;
   555→  let out = "";
   556→  let complete = false;
   557→  while (i < buffer.length) {
   558→    const ch = buffer[i];
   559→    if (ch === "\\") {
   560→      if (i + 1 >= buffer.length) break;
   561→      const next = buffer[i + 1];
   562→      if (next === "n") out += "\n";
   563→      else if (next === "t") out += "\t";
   564→      else if (next === "r") out += "\r";
   565→      else if (next === '"') out += '"';
   566→      else if (next === "\\") out += "\\";
   567→      else if (next === "/") out += "/";
   568→      else if (next === "b") out += "\b";
   569→      else if (next === "f") out += "\f";
   570→      else if (next === "u") {
   571→        if (i + 5 >= buffer.length) break;
   572→        const hex = buffer.slice(i + 2, i + 6);
   573→        try {
   574→          out += String.fromCharCode(parseInt(hex, 16));
   575→          i += 4;
   576→        } catch {
   577→          out += next;
   578→        }
   579→      } else {
   580→        out += next;
   581→      }
   582→      i += 2;
   583→    } else if (ch === '"') {
   584→      complete = true;
   585→      break;
   586→    } else {
   587→      out += ch;
   588→      i++;
   589→    }
   590→  }
   591→  return { text: out, complete };
   592→}

   594→// Parse an SSE stream (text/event-stream) and yield content deltas.
   595→// Tolerates \n\n and \r\n\r\n separators; ignores keep-alive lines.
   596→async function* parseSSEStream(
   597→  stream: ReadableStream<Uint8Array> | NodeJS.ReadableStream | any
   598→): AsyncGenerator<string> {
   599→  const reader = (stream as any).getReader?.();
   600→  if (!reader) {
   601→    // Node stream fallback
   602→    const iter = (stream as any)[Symbol.asyncIterator]?.();
   603→    if (iter) {
   604→      const decoder = new TextDecoder();
   605→      let buffer = "";
   606→      while (true) {
   607→        const { value, done } = await iter.next();
   608→        if (done) break;
   609→        buffer += decoder.decode(value, { stream: true });
   610→        let idx;
   611→        while ((idx = buffer.indexOf("\n\n")) !== -1) {
   612→          const chunk = buffer.slice(0, idx);
   613→          buffer = buffer.slice(idx + 2);
   614→          for (const line of chunk.split("\n")) {
   615→            const m = line.match(/^data:\s?(.*)$/);
   616→            if (!m) continue;
   617→            const data = m[1];
   618→            if (data === "[DONE]") return;
   619→            try {
   620→              const json = JSON.parse(data);
   621→              const delta = json?.choices?.[0]?.delta?.content;
   622→              if (typeof delta === "string" && delta) yield delta;
   623→            } catch {
   624→              // skip malformed
   625→            }
   626→          }
   627→        }
   628→      }
   629→    }
   630→    return;
   631→  }
   632→  const decoder = new TextDecoder();
   633→  let buffer = "";
   634→  while (true) {
   635→    const { value, done } = await reader.read();
   636→    if (done) break;
   637→    buffer += decoder.decode(value, { stream: true });
   638→    let idx;
   639→    while ((idx = buffer.indexOf("\n\n")) !== -1) {
   640→      const chunk = buffer.slice(0, idx);
   641→      buffer = buffer.slice(idx + 2);
   642→      for (const line of chunk.split("\n")) {
   643→        const m = line.match(/^data:\s?(.*)$/);
   644→        if (!m) continue;
   645→        const data = m[1];
   646→        if (data === "[DONE]") return;
   647→        try {
   648→          const json = JSON.parse(data);
   649→          const delta = json?.choices?.[0]?.delta?.content;
   650→          if (typeof delta === "string" && delta) yield delta;
   651→        } catch {
   652→          // skip malformed
   653→        }
   654→      }
   655→    }
   656→  }
   657→}

   659→export async function generateArcadeTurnStreaming(
   660→  opts: {
   661→    mode: ArcadeMode;
   662→    dream: { rawText: string; analysis: any };
   663→    state: SimulationState;
   664→    history: { userAction: string; sceneText: string }[];
   665→    userAction: string;
   666→    dreamMotifs: string[];
   667→  },
   668→  onDelta: (delta: string) => void
   669→): Promise<{ response: ArcadeTurnResponse; raw: string }> {
   670→  const prompt = ARCADE_SYSTEM_PROMPT(opts);
   671→  const client = await zai();
   672→
   673→  let raw = "";
   674→  try {
   675→    const completion = await client.chat.completions.create({
   676→      messages: [
   677→        { role: "assistant", content: prompt.system },
   678→        { role: "user", content: prompt.user },
   679→      ],
   680→      stream: true,
   681→      thinking: { type: "disabled" },
   682→    });
   683→
   684→    // The SDK returns response.body (ReadableStream) when stream:true and
   685→    // text/event-stream; otherwise returns parsed JSON.
   686→    const stream =
   687→      completion && (completion.body instanceof ReadableStream
   688→        ? completion.body
   689→        : completion instanceof ReadableStream
   690→        ? completion
   691→        : null);
   692→
   693→    if (stream) {
   694→      let sentText = "";
   695→      for await (const delta of parseSSEStream(stream)) {
   696→        raw += delta;
   697→        const { text } = extractPartialSceneText(raw);
   698→        if (text.length > sentText.length) {
   699→          // emit only the newly completed slice
   700→          onDelta(text.slice(sentText.length));
   701→          sentText = text;
   702→        }
   703→      }
   704→    } else {
   705→      // non-streaming fallback (server didn't honour stream:true)
   706→      raw = completion?.choices?.[0]?.message?.content ?? "";
   707→      const { text } = extractPartialSceneText(raw);
   708→      if (text) onDelta(text);
   709→    }
   710→  } catch (e) {
   711→    console.warn("[ai] arcade streaming turn failed:", e);
   712→    throw e;
   713→  }
   714→
   715→  let parsed = extractJSON(raw);
   716→  if (!parsed) {
   717→    // One repair attempt with non-streaming
   718→    console.warn("[ai] streaming arcade turn JSON unparseable — retrying once. head:", brief(raw));
   719→    try {
   720→      const retry = await client.chat.completions.create({
   721→        messages: [
   722→          { role: "assistant", content: prompt.system },
   723→          { role: "user", content: prompt.user },
   724→          { role: "assistant", content: raw.slice(0, 6000) },
   725→          {
   726→            role: "user",
   727→            content:
   728→              "Your previous response was not valid JSON. Return ONLY the JSON object matching the schema — no prose, no code fences. Keep sceneText vivid but finish the object.",
   729→          },
   730→        ],
   731→        thinking: { type: "disabled" },
   732→      });
   733→      raw = retry?.choices?.[0]?.message?.content ?? "";
   734→      parsed = extractJSON(raw);
   735→      if (parsed) {
   736→        // emit the repaired sceneText in one chunk so the client catches up
   737→        const { text } = extractPartialSceneText(raw);
   738→        if (text) onDelta(text);
   739→      }
   740→    } catch (e) {
   741→      console.warn("[ai] streaming arcade turn repair attempt failed:", e);
   742→    }
   743→  }
   744→  if (!parsed) {
   745→    console.warn("[ai] streaming arcade turn fallback engaged. head:", brief(raw));
   746→    const fallback: ArcadeTurnResponse = {
   747→      sceneText:
   748→        "The dream stutters for a moment — the scene is still forming. Try describing what you do next.",
   749→      choices: [
   750→        { id: "continue", label: "Press onward into the dream", hint: "Move deeper" },
   751→        { id: "observe", label: "Pause and observe your surroundings" },
   752→      ],
   753→      proposedDelta: { reasoning: "Model response was malformed; no state change applied." },
   754→      discoveredMotifs: [],
   755→    };
   756→    onDelta(fallback.sceneText);
   757→    return { response: fallback, raw };
   758→  }
   759→
   760→  const safe = turnSchema.parse(parsed);
   761→  const response = buildArcadeResponse(safe);
   762→  return { response, raw };
   763→}
