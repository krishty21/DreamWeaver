    25→    25→function clamp(n: unknown, lo = 0, hi = 1): number {
    26→    26→  const v = typeof n === "number" ? n : Number(n);
    27→    27→  if (!isFinite(v)) return lo;
    28→    28→  return Math.max(lo, Math.min(hi, v));
    29→    29→}

    31→    31→// Normalize common model quirks before parsing: smart quotes, // and /* */
    32→    32→// comments, BOM, zero-width chars.
    33→    33→function normalizeJSONSource(t: string): string {
    34→    34→  return t
    35→    35→    .replace(/\uFEFF/g, "")
    36→    36→    .replace(/[\u200B-\u200D\u2060]/g, "")
    37→    37→    .replace(/[\u201C\u201D]/g, '"')
    38→    38→    .replace(/[\u2018\u2019]/g, "'")
    39→    39→    .replace(/[\u2013\u2014]/g, "-")
    40→    40→    // line comments (only outside strings — the crude strip below is safe
    41→    41→    // enough for model output where // almost always starts a comment)
    42→    42→    .replace(/^\s*\/\/.*$/gm, "")
    43→    43→    .replace(/\/\*[\s\S]*?\*\//g, "");
    44→    44→}

    46→    46→// Attempt JSON.parse with progressively more aggressive repairs.
    47→    47→function parseWithRepairs(s: string): any | null {
    48→    48→  // 1. direct parse
    49→    49→  try {
    50→    50→    return JSON.parse(s);
    51→    51→  } catch {}
    52→    52→
    53→    53→  // 2. trailing commas
    54→    54→  try {
    55→    55→    return JSON.parse(s.replace(/,\s*([}\]])/g, "$1"));
    56→    56→  } catch {}
    57→    57→
    58→    58→  // 3. truncated output — cut back to the last value that parsed cleanly,
    59→    59→  //    then close any open brackets/braces.
    60→    60→  try {
    61→    61→    const repaired = repairTruncated(s.replace(/,\s*([}\]])/g, "$1"));
    62→    62→    if (repaired) return JSON.parse(repaired);
    63→    63→  } catch {}
    64→    64→
    65→    65→  return null;
    66→    66→}

    68→    68→// Repair a truncated JSON document: terminate a dangling string, strip the
    69→    69→// tail after the last complete value if possible, then close open structures.
    70→    70→function repairTruncated(s: string): string | null {
    71→    71→  // Walk the string tracking string-state and open brackets.
    72→    72→  let inString = false;
    73→    73→  let escape = false;
    74→    74→  const stack: string[] = [];
    75→    75→  let lastSafe = -1; // index of the last "clean value boundary" outside strings
    76→    76→  for (let i = 0; i < s.length; i++) {
    77→    77→    const ch = s[i];
    78→    78→    if (escape) {
    79→    79→      escape = false;
    80→    80→      continue;
    81→    81→    }
    82→    82→    if (ch === "\\" && inString) {
    83→    83→      escape = true;
    84→    84→      continue;
    85→    85→    }
    86→    86→    if (ch === '"') {
    87→    87→      inString = !inString;
    88→    88→      continue;
    89→    89→    }
    90→    90→    if (inString) continue;
    91→    91→    if (ch === "{" || ch === "[") stack.push(ch);
    92→    92→    else if (ch === "}" || ch === "]") stack.pop();
    93→    93→    else if (ch === "," && stack.length > 0) lastSafe = i;
    94→    94→  }
    95→    95→  if (stack.length === 0) return null; // nothing to close — not truncation-shaped
    96→    96→
    97→    97→  // Cut at the last comma that sat inside an open structure (drops the
    98→    98→  // incomplete trailing value), unless that would discard the only content.
    99→    99→  let out: string;
   100→   100→  if (lastSafe > 0) {
   101→   101→    out = s.slice(0, lastSafe).replace(/[,\s]+$/, "");
   102→   102→  } else {
   103→   103→    // No safe comma boundary — keep everything; if we're mid-string,
   104→   104→    // terminate the string so the document can still be closed.
   105→   105→    out = s.replace(/[,\s]+$/, "");
   106→   106→    if (inString) out += '"';
   107→   107→  }
   108→   108→
   109→   109→  // Recompute the open-structure stack on the trimmed source.
   110→   110→  let inStr2 = false;
   111→   111→  let esc2 = false;
   112→   112→  const stack2: string[] = [];
   113→   113→  for (let i = 0; i < out.length; i++) {
   114→   114→    const ch = out[i];
   115→   115→    if (esc2) {
   116→   116→      esc2 = false;
   117→   117→      continue;
   118→   118→    }
   119→   119→    if (ch === "\\" && inStr2) {
   120→   120→      esc2 = true;
   121→   121→      continue;
   122→   122→    }
   123→   123→    if (ch === '"') {
   124→   124→      inStr2 = !inStr2;
   125→   125→      continue;
   126→   126→    }
   127→   127→    if (inStr2) continue;
   128→   128→    if (ch === "{" || ch === "[") stack2.push(ch);
   129→   129→    else if (ch === "}" || ch === "]") stack2.pop();
   130→   130→  }
   131→   131→  if (inStr2) return null; // still malformed (e.g. stray quote) — give up
   132→   132→  while (stack2.length > 0) {
   133→   133→    const open = stack2.pop();
   134→   134→    out += open === "{" ? "}" : "]";
   135→   135→  }
   136→   136→  return out;
   137→   137→}

   139→   139→// Try to extract a JSON object from a model response that may contain
   140→   140→// surrounding prose / code fences / truncation.
   141→   141→function extractJSON(text: string): any | null {
   142→   142→  if (!text) return null;
   143→   143→  // strip code fences
   144→   144→  let t = text.trim();
   145→   145→  const fence = t.match(/```(?:json)?\s*([\s\S]*?)```/i);
   146→   146→  if (fence) t = fence[1].trim();
   147→   147→  t = normalizeJSONSource(t);
   148→   148→
   149→   149→  const firstBrace = t.indexOf("{");
   150→   150→  const firstBracket = t.indexOf("[");
   151→   151→  // A top-level array starts before any object — try the array first.
   152→   152→  const arrayFirst =
   153→   153→    firstBracket !== -1 && (firstBrace === -1 || firstBracket < firstBrace);
   154→   154→  if (arrayFirst) {
   155→   155→    const lastBracket = t.lastIndexOf("]");
   156→   156→    const candidate =
   157→   157→      lastBracket > firstBracket
   158→   158→        ? t.slice(firstBracket, lastBracket + 1) // possibly complete array
   159→   159→        : t.slice(firstBracket); // truncated array (final ] dropped)
   160→   160→    const parsedArr = parseWithRepairs(candidate);
   161→   161→    if (parsedArr !== null) return parsedArr;
   162→   162→  }
   163→   163→
   164→   164→  // Object extraction: first { ... last }, with truncation fallbacks.
   165→   165→  const first = firstBrace;
   166→   166→  const last = t.lastIndexOf("}");
   167→   167→  if (first === -1) return null;
   168→   168→  if (last > first) {
   169→   169→    const parsed = parseWithRepairs(t.slice(first, last + 1));
   170→   170→    if (parsed !== null) return parsed;
   171→   171→  }
   172→   172→  // Last resort: everything after the first { is a truncated object.
   173→   173→  return parseWithRepairs(t.slice(first));
   174→   174→}

   542→   542→//
   543→   543→// Streams the model's sceneText to the client as it is produced. The SDK
   544→   544→// returns the raw fetch Response.body (a ReadableStream) when stream:true
   545→   545→// is requested AND the server responds with text/event-stream; otherwise it
   546→   546→// returns the parsed JSON (we fall back to a single delta emit).
   547→   547→//
   548→   548→// The model emits the response as a JSON object whose first field is
   549→   549→// `sceneText`. As chunks arrive we incrementally extract the partial
   550→   550→// sceneText from the partial JSON buffer (handling escapes) and emit each
   551→   551→// newly-completed slice via onDelta. The complete JSON is parsed at end
   552→   552→// and the structured response returned, exactly like generateArcadeTurn.

   558→   558→      id: c.id || `choice-${i + 1}`,
   559→   559→      label: String(c.label).slice(0, 120),
   560→   560→      hint: c.hint ? String(c.hint).slice(0, 200) : undefined,
   561→   561→    })),
   562→   562→    proposedDelta: clampDelta(safe.proposedDelta),
   563→   563→    discoveredMotifs: (safe.proposedDelta.discoveredMotifs || [])
   564→   564→      .map((s) => String(s).slice(0, 60).toLowerCase())
   565→   565→      .slice(0, 8),
   566→   566→  };
   567→   567→}

   569→   569→// Extract the partial sceneText value from a partial JSON buffer.
   570→   570→// Returns { text, complete } where complete=true if the closing quote was seen.
   571→   571→function extractPartialSceneText(buffer: string): { text: string; complete: boolean } {
   572→   572→  const key = `"sceneText"`;
   573→   573→  const idx = buffer.indexOf(key);
   574→   574→  if (idx === -1) return { text: "", complete: false };
   575→   575→
   576→   576→  // skip whitespace
   577→   577→  let i = idx + key.length;
   578→   578→  while (i < buffer.length && /\s/.test(buffer[i])) i++;
   579→   579→  if (i >= buffer.length || buffer[i] !== ":") return { text: "", complete: false };
   580→   580→  i++;
   581→   581→  while (i < buffer.length && /\s/.test(buffer[i])) i++;
   582→   582→  if (i >= buffer.length) return { text: "", complete: false };
   583→   583→  if (buffer[i] !== '"') return { text: "", complete: false };
   584→   584→
   585→   585→  // string content
   586→   586→  i++;
   587→   587→  let out = "";
   588→   588→  let complete = false;
   589→   589→  while (i < buffer.length) {
   590→   590→    const ch = buffer[i];
   591→   591→    if (ch === "\\") {
   592→   592→      if (i + 1 >= buffer.length) break;
   593→   593→      const next = buffer[i + 1];
   594→   594→      if (next === "n") out += "\n";
   595→   595→      else if (next === "t") out += "\t";
   596→   596→      else if (next === "r") out += "\r";
   597→   597→      else if (next === '"') out += '"';
   598→   598→      else if (next === "\\") out += "\\";
   599→   599→      else if (next === "/") out += "/";
   600→   600→      else if (next === "b") out += "\b";
   601→   601→      else if (next === "f") out += "\f";
   602→   602→      else if (next === "u") {
   603→   603→        if (i + 5 >= buffer.length) break;
   604→   604→        const hex = buffer.slice(i + 2, i + 6);
   605→   605→        try {
   606→   606→          out += String.fromCharCode(parseInt(hex, 16));
   607→   607→          i += 4;
   608→   608→        } catch {
   609→   609→          out += next;
   610→   610→        }
   611→   611→      } else {
   612→   612→        out += next;
   613→   613→      }
   614→   614→      i += 2;
   615→   615→    } else if (ch === '"') {
   616→   616→      complete = true;
   617→   617→      break;
   618→   618→    } else {
   619→   619→      out += ch;
   620→   620→      i++;
   621→   621→    }
   622→   622→  }
   623→   623→  return { text: out, complete };
   624→   624→}

   626→   626→// Parse an SSE stream (text/event-stream) and yield content deltas.
   627→   627→// Tolerates \n\n and \r\n\r\n separators; ignores keep-alive lines.
   628→   628→async function* parseSSEStream(
   629→   629→  stream: ReadableStream<Uint8Array> | NodeJS.ReadableStream | any
   630→   630→): AsyncGenerator<string> {
   631→   631→  const reader = (stream as any).getReader?.();
   632→   632→  if (!reader) {
   633→   633→    // Node stream fallback
   634→   634→    const iter = (stream as any)[Symbol.asyncIterator]?.();
   635→   635→    if (iter) {
   636→   636→      const decoder = new TextDecoder();
   637→   637→      let buffer = "";
   638→   638→      while (true) {
   639→   639→        const { value, done } = await iter.next();
   640→   640→        if (done) break;
   641→   641→        buffer += decoder.decode(value, { stream: true });
   642→   642→        let idx;
   643→   643→        while ((idx = buffer.indexOf("\n\n")) !== -1) {
   644→   644→          const chunk = buffer.slice(0, idx);
   645→   645→          buffer = buffer.slice(idx + 2);
   646→   646→          for (const line of chunk.split("\n")) {
   647→   647→            const m = line.match(/^data:\s?(.*)$/);
   648→   648→            if (!m) continue;
   649→   649→            const data = m[1];
   650→   650→            if (data === "[DONE]") return;
   651→   651→            try {
   652→   652→              const json = JSON.parse(data);
   653→   653→              const delta = json?.choices?.[0]?.delta?.content;
   654→   654→              if (typeof delta === "string" && delta) yield delta;
   655→   655→            } catch {
   656→   656→              // skip malformed
   657→   657→            }
   658→   658→          }
   659→   659→        }
   660→   660→      }
   661→   661→    }
   662→   662→    return;
   663→   663→  }
   664→   664→  const decoder = new TextDecoder();
   665→   665→  let buffer = "";
   666→   666→  while (true) {
   667→   667→    const { value, done } = await reader.read();
   668→   668→    if (done) break;
   669→   669→    buffer += decoder.decode(value, { stream: true });
   670→   670→    let idx;
   671→   671→    while ((idx = buffer.indexOf("\n\n")) !== -1) {
   672→   672→      const chunk = buffer.slice(0, idx);
   673→   673→      buffer = buffer.slice(idx + 2);
   674→   674→      for (const line of chunk.split("\n")) {
   675→   675→        const m = line.match(/^data:\s?(.*)$/);
   676→   676→        if (!m) continue;
   677→   677→        const data = m[1];
   678→   678→        if (data === "[DONE]") return;
   679→   679→        try {
   680→   680→          const json = JSON.parse(data);
   681→   681→          const delta = json?.choices?.[0]?.delta?.content;
   682→   682→          if (typeof delta === "string" && delta) yield delta;
   683→   683→        } catch {
   684→   684→          // skip malformed
   685→   685→        }
   686→   686→      }
   687→   687→    }
   688→   688→  }
   689→   689→}

   705→   705→  const client = await zai();

   711→   711→        { role: "assistant", content: prompt.system },
   712→   712→        { role: "user", content: prompt.user },

   719→   719→    // text/event-stream; otherwise returns parsed JSON.

   721→   721→      completion && (completion.body instanceof ReadableStream
   722→   722→        ? completion.body
   723→   723→        : completion instanceof ReadableStream

   747→   747→  }

   749→   749→  let parsed = extractJSON(raw);
   750→   750→  if (!parsed) {
   751→   751→    // One repair attempt with non-streaming
   752→   752→    console.warn("[ai] streaming arcade turn JSON unparseable — retrying once. head:", brief(raw));
   753→   753→    try {

   755→   755→        messages: [

   763→   763→          },

   768→   768→      parsed = extractJSON(raw);
   769→   769→      if (parsed) {
   770→   770→        // emit the repaired sceneText in one chunk so the client catches up
   771→   771→        const { text } = extractPartialSceneText(raw);
   772→   772→        if (text) onDelta(text);
   773→   773→      }
