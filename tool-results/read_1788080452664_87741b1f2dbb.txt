    25→function clamp(n: unknown, lo = 0, hi = 1): number {
    26→  const v = typeof n === "number" ? n : Number(n);
    27→  if (!isFinite(v)) return lo;
    28→  return Math.max(lo, Math.min(hi, v));
    29→}

    31→// Normalize common model quirks before parsing: smart quotes, // and /* */
    32→// comments, BOM, zero-width chars.
    33→function normalizeJSONSource(t: string): string {
    34→  return t
    35→    .replace(/\uFEFF/g, "")
    36→    .replace(/[\u200B-\u200D\u2060]/g, "")
    37→    .replace(/[\u201C\u201D]/g, '"')
    38→    .replace(/[\u2018\u2019]/g, "'")
    39→    .replace(/[\u2013\u2014]/g, "-")
    40→    // line comments (only outside strings — the crude strip below is safe
    41→    // enough for model output where // almost always starts a comment)
    42→    .replace(/^\s*\/\/.*$/gm, "")
    43→    .replace(/\/\*[\s\S]*?\*\//g, "");
    44→}

    46→// Attempt JSON.parse with progressively more aggressive repairs.
    47→function parseWithRepairs(s: string): any | null {
    48→  // 1. direct parse
    49→  try {
    50→    return JSON.parse(s);
    51→  } catch {}
    52→
    53→  // 2. trailing commas
    54→  try {
    55→    return JSON.parse(s.replace(/,\s*([}\]])/g, "$1"));
    56→  } catch {}
    57→
    58→  // 3. truncated output — cut back to the last value that parsed cleanly,
    59→  //    then close any open brackets/braces.
    60→  try {
    61→    const repaired = repairTruncated(s.replace(/,\s*([}\]])/g, "$1"));
    62→    if (repaired) return JSON.parse(repaired);
    63→  } catch {}
    64→
    65→  return null;
    66→}

    68→// Repair a truncated JSON document: terminate a dangling string, strip the
    69→// tail after the last complete value if possible, then close open structures.
    70→function repairTruncated(s: string): string | null {
    71→  // Walk the string tracking string-state and open brackets.
    72→  let inString = false;
    73→  let escape = false;
    74→  const stack: string[] = [];
    75→  let lastSafe = -1; // index of the last "clean value boundary" outside strings
    76→  for (let i = 0; i < s.length; i++) {
    77→    const ch = s[i];
    78→    if (escape) {
    79→      escape = false;
    80→      continue;
    81→    }
    82→    if (ch === "\\" && inString) {
    83→      escape = true;
    84→      continue;
    85→    }
    86→    if (ch === '"') {
    87→      inString = !inString;
    88→      continue;
    89→    }
    90→    if (inString) continue;
    91→    if (ch === "{" || ch === "[") stack.push(ch);
    92→    else if (ch === "}" || ch === "]") stack.pop();
    93→    else if (ch === "," && stack.length > 0) lastSafe = i;
    94→  }
    95→  if (stack.length === 0) return null; // nothing to close — not truncation-shaped
    96→
    97→  // Cut at the last comma that sat inside an open structure (drops the
    98→  // incomplete trailing value), unless that would discard the only content.
    99→  let out: string;
   100→  if (lastSafe > 0) {
   101→    out = s.slice(0, lastSafe).replace(/[,\s]+$/, "");
   102→  } else {
   103→    // No safe comma boundary — keep everything; if we're mid-string,
   104→    // terminate the string so the document can still be closed.
   105→    out = s.replace(/[,\s]+$/, "");
   106→    if (inString) out += '"';
   107→  }
   108→
   109→  // Recompute the open-structure stack on the trimmed source.
   110→  let inStr2 = false;
   111→  let esc2 = false;
   112→  const stack2: string[] = [];
   113→  for (let i = 0; i < out.length; i++) {
   114→    const ch = out[i];
   115→    if (esc2) {
   116→      esc2 = false;
   117→      continue;
   118→    }
   119→    if (ch === "\\" && inStr2) {
   120→      esc2 = true;
   121→      continue;
   122→    }
   123→    if (ch === '"') {
   124→      inStr2 = !inStr2;
   125→      continue;
   126→    }
   127→    if (inStr2) continue;
   128→    if (ch === "{" || ch === "[") stack2.push(ch);
   129→    else if (ch === "}" || ch === "]") stack2.pop();
   130→  }
   131→  if (inStr2) return null; // still malformed (e.g. stray quote) — give up
   132→  while (stack2.length > 0) {
   133→    const open = stack2.pop();
   134→    out += open === "{" ? "}" : "]";
   135→  }
   136→  return out;
   137→}

   139→// Try to extract a JSON object from a model response that may contain
   140→// surrounding prose / code fences / truncation.
   141→function extractJSON(text: string): any | null {
   142→  if (!text) return null;
   143→  // strip code fences
   144→  let t = text.trim();
   145→  const fence = t.match(/```(?:json)?\s*([\s\S]*?)```/i);
   146→  if (fence) t = fence[1].trim();
   147→  t = normalizeJSONSource(t);
   148→
   149→  const firstBrace = t.indexOf("{");
   150→  const firstBracket = t.indexOf("[");
   151→  // A top-level array starts before any object — try the array first.
   152→  const arrayFirst =
   153→    firstBracket !== -1 && (firstBrace === -1 || firstBracket < firstBrace);
   154→  if (arrayFirst) {
   155→    const lastBracket = t.lastIndexOf("]");
   156→    const candidate =
   157→      lastBracket > firstBracket
   158→        ? t.slice(firstBracket, lastBracket + 1) // possibly complete array
   159→        : t.slice(firstBracket); // truncated array (final ] dropped)
   160→    const parsedArr = parseWithRepairs(candidate);
   161→    if (parsedArr !== null) return parsedArr;
   162→  }
   163→
   164→  // Object extraction: first { ... last }, with truncation fallbacks.
   165→  const first = firstBrace;
   166→  const last = t.lastIndexOf("}");
   167→  if (first === -1) return null;
   168→  if (last > first) {
   169→    const parsed = parseWithRepairs(t.slice(first, last + 1));
   170→    if (parsed !== null) return parsed;
   171→  }
   172→  // Last resort: everything after the first { is a truncated object.
   173→  return parseWithRepairs(t.slice(first));
   174→}

   252→function brief(raw: string): string {
   253→  const s = JSON.stringify(raw);
   254→  if (s.length <= 700) return s;
   255→  // head + tail: truncation diagnosis needs the end of the response
   256→  return `${s.slice(0, 400)} …[len ${s.length}]… ${s.slice(-260)}`;
   257→}

   511→function clampDelta(d: any): ProposedDelta {
   512→  const out: ProposedDelta = {};
   513→  if (typeof d.fear === "number") out.fear = clamp(d.fear, 0, 100);
   514→  if (typeof d.lucidity === "number") out.lucidity = clamp(d.lucidity, 0, 100);
   515→  if (typeof d.stability === "number") out.stability = clamp(d.stability, 0, 100);
   516→  if (typeof d.agency === "number") out.agency = clamp(d.agency, 0, 100);
   517→  if (Array.isArray(d.discoveredMotifs))
   518→    out.discoveredMotifs = d.discoveredMotifs.map((s: any) => String(s).slice(0, 60).toLowerCase()).slice(0, 8);
   519→  if (typeof d.visitedScene === "string") out.visitedScene = String(d.visitedScene).slice(0, 80);
   520→  if (Array.isArray(d.inventoryAdd))
   521→    out.inventoryAdd = d.inventoryAdd.map((s: any) => String(s).slice(0, 60)).slice(0, 5);
   522→  if (
   523→    d.phase === "opening" ||
   524→    d.phase === "developing" ||
   525→    d.phase === "climax" ||
   526→    d.phase === "resolving"
   527→  )
   528→    out.phase = d.phase;
   529→  if (
   530→    d.ending === "collapse" ||
   531→    d.ending === "escape" ||
   532→    d.ending === "control" ||
   533→    d.ending === "unresolved" ||
   534→    d.ending === "transformed"
   535→  )
   536→    out.ending = d.ending;
   537→  if (typeof d.reasoning === "string") out.reasoning = String(d.reasoning).slice(0, 400);
   538→  return out;
   539→}

   544→// returns the raw fetch Response.body (a ReadableStream) when stream:true
   545→// is requested AND the server responds with text/event-stream; otherwise it
   546→// returns the parsed JSON (we fall back to a single delta emit).
   547→//
   548→// The model emits the response as a JSON object whose first field is
   549→// `sceneText`. As chunks arrive we incrementally extract the partial
   550→// sceneText from the partial JSON buffer (handling escapes) and emit each
   551→// newly-completed slice via onDelta. The complete JSON is parsed at end
   552→// and the structured response returned, exactly like generateArcadeTurn.
   553→
   554→function buildArcadeResponse(safe: z.infer<typeof turnSchema>): ArcadeTurnResponse {
   555→  return {
   556→    sceneText: String(safe.sceneText).slice(0, 2000),
   557→    choices: safe.choices.map((c, i) => ({
   558→      id: c.id || `choice-${i + 1}`,
   559→      label: String(c.label).slice(0, 120),
   560→      hint: c.hint ? String(c.hint).slice(0, 200) : undefined,
   561→    })),
   562→    proposedDelta: clampDelta(safe.proposedDelta),
   563→    discoveredMotifs: (safe.proposedDelta.discoveredMotifs || [])
   564→      .map((s) => String(s).slice(0, 60).toLowerCase())
   565→      .slice(0, 8),
   566→  };
   567→}
   568→
   569→// Extract the partial sceneText value from a partial JSON buffer.
   570→// Returns { text, complete } where complete=true if the closing quote was seen.
   571→function extractPartialSceneText(buffer: string): { text: string; complete: boolean } {
   572→  const key = `"sceneText"`;
   573→  const idx = buffer.indexOf(key);
   574→  if (idx === -1) return { text: "", complete: false };
   575→
   576→  // skip whitespace
   577→  let i = idx + key.length;
   578→  while (i < buffer.length && /\s/.test(buffer[i])) i++;
   579→  if (i >= buffer.length || buffer[i] !== ":") return { text: "", complete: false };
   580→  i++;
   581→  while (i < buffer.length && /\s/.test(buffer[i])) i++;
   582→  if (i >= buffer.length) return { text: "", complete: false };
   583→  if (buffer[i] !== '"') return { text: "", complete: false };
   584→
   585→  // string content
   586→  i++;
   587→  let out = "";
   588→  let complete = false;
   589→  while (i < buffer.length) {
   590→    const ch = buffer[i];
   591→    if (ch === "\\") {
   592→      if (i + 1 >= buffer.length) break;
   593→      const next = buffer[i + 1];
   594→      if (next === "n") out += "\n";
   595→      else if (next === "t") out += "\t";
   596→      else if (next === "r") out += "\r";
   597→      else if (next === '"') out += '"';
   598→      else if (next === "\\") out += "\\";
   599→      else if (next === "/") out += "/";
   600→      else if (next === "b") out += "\b";
   601→      else if (next === "f") out += "\f";
   602→      else if (next === "u") {
   603→        if (i + 5 >= buffer.length) break;
   604→        const hex = buffer.slice(i + 2, i + 6);
   605→        try {
   606→          out += String.fromCharCode(parseInt(hex, 16));
   607→          i += 4;
   608→        } catch {
   609→          out += next;
   610→        }
   611→      } else {
   612→        out += next;
   613→      }
   614→      i += 2;
   615→    } else if (ch === '"') {
   616→      complete = true;
   617→      break;
   618→    } else {
   619→      out += ch;
   620→      i++;
   621→    }
   622→  }
   623→  return { text: out, complete };
   624→}

   626→// Parse an SSE stream (text/event-stream) and yield content deltas.
   627→// Tolerates \n\n and \r\n\r\n separators; ignores keep-alive lines.
   628→async function* parseSSEStream(
   629→  stream: ReadableStream<Uint8Array> | NodeJS.ReadableStream | any
   630→): AsyncGenerator<string> {
   631→  const reader = (stream as any).getReader?.();
   632→  if (!reader) {
   633→    // Node stream fallback
   634→    const iter = (stream as any)[Symbol.asyncIterator]?.();
   635→    if (iter) {
   636→      const decoder = new TextDecoder();
   637→      let buffer = "";
   638→      while (true) {
   639→        const { value, done } = await iter.next();
   640→        if (done) break;
   641→        buffer += decoder.decode(value, { stream: true });
   642→        let idx;
   643→        while ((idx = buffer.indexOf("\n\n")) !== -1) {
   644→          const chunk = buffer.slice(0, idx);
   645→          buffer = buffer.slice(idx + 2);
   646→          for (const line of chunk.split("\n")) {
   647→            const m = line.match(/^data:\s?(.*)$/);
   648→            if (!m) continue;
   649→            const data = m[1];
   650→            if (data === "[DONE]") return;
   651→            try {
   652→              const json = JSON.parse(data);
   653→              const delta = json?.choices?.[0]?.delta?.content;
   654→              if (typeof delta === "string" && delta) yield delta;
   655→            } catch {
   656→              // skip malformed
   657→            }
   658→          }
   659→        }
   660→      }
   661→    }
   662→    return;
   663→  }
   664→  const decoder = new TextDecoder();
   665→  let buffer = "";
   666→  while (true) {
   667→    const { value, done } = await reader.read();
   668→    if (done) break;
   669→    buffer += decoder.decode(value, { stream: true });
   670→    let idx;
   671→    while ((idx = buffer.indexOf("\n\n")) !== -1) {
   672→      const chunk = buffer.slice(0, idx);
   673→      buffer = buffer.slice(idx + 2);
   674→      for (const line of chunk.split("\n")) {
   675→        const m = line.match(/^data:\s?(.*)$/);
   676→        if (!m) continue;
   677→        const data = m[1];
   678→        if (data === "[DONE]") return;
   679→        try {
   680→          const json = JSON.parse(data);
   681→          const delta = json?.choices?.[0]?.delta?.content;
   682→          if (typeof delta === "string" && delta) yield delta;
   683→        } catch {
   684→          // skip malformed
   685→        }
   686→      }
   687→    }
   688→  }
   689→}

   691→export async function generateArcadeTurnStreaming(
   692→  opts: {
   693→    mode: ArcadeMode;
   694→    dream: { rawText: string; analysis: any };
   695→    state: SimulationState;
   696→    history: { userAction: string; sceneText: string }[];
   697→    userAction: string;
   698→    dreamMotifs: string[];
   699→    dreamLaws?: { law: string; evidence?: string }[];
   700→    historicalConnections?: { motif: string; priorDreamCount: number }[];
   701→  },
   702→  onDelta: (delta: string) => void
   703→): Promise<{ response: ArcadeTurnResponse; raw: string }> {
   704→  const prompt = ARCADE_SYSTEM_PROMPT(opts);
   705→  const client = await zai();
   706→
   707→  let raw = "";
   708→  try {
   709→    const completion = await client.chat.completions.create({
   710→      messages: [
   711→        { role: "assistant", content: prompt.system },
   712→        { role: "user", content: prompt.user },
   713→      ],
   714→      stream: true,
   715→      thinking: { type: "disabled" },
   716→    });
   717→
   718→    // The SDK returns response.body (ReadableStream) when stream:true and
   719→    // text/event-stream; otherwise returns parsed JSON.
   720→    const stream =
   721→      completion && (completion.body instanceof ReadableStream
   722→        ? completion.body
   723→        : completion instanceof ReadableStream
   724→        ? completion
   725→        : null);
   726→
   727→    if (stream) {
   728→      let sentText = "";
   729→      for await (const delta of parseSSEStream(stream)) {
   730→        raw += delta;
   731→        const { text } = extractPartialSceneText(raw);
   732→        if (text.length > sentText.length) {
   733→          // emit only the newly completed slice
   734→          onDelta(text.slice(sentText.length));
   735→          sentText = text;
   736→        }
   737→      }
   738→    } else {
   739→      // non-streaming fallback (server didn't honour stream:true)
   740→      raw = completion?.choices?.[0]?.message?.content ?? "";
   741→      const { text } = extractPartialSceneText(raw);
   742→      if (text) onDelta(text);
   743→    }
   744→  } catch (e) {
   745→    console.warn("[ai] arcade streaming turn failed:", e);
   746→    throw e;
   747→  }
   748→
   749→  let parsed = extractJSON(raw);
   750→  if (!parsed) {
   751→    // One repair attempt with non-streaming
   752→    console.warn("[ai] streaming arcade turn JSON unparseable — retrying once. head:", brief(raw));
   753→    try {
   754→      const retry = await client.chat.completions.create({
   755→        messages: [
   756→          { role: "assistant", content: prompt.system },
   757→          { role: "user", content: prompt.user },
   758→          { role: "assistant", content: raw.slice(0, 6000) },
   759→          {
   760→            role: "user",
   761→            content:
   762→              "Your previous response was not valid JSON. Return ONLY the JSON object matching the schema — no prose, no code fences. Keep sceneText vivid but finish the object.",
   763→          },
   764→        ],
   765→        thinking: { type: "disabled" },
   766→      });
   767→      raw = retry?.choices?.[0]?.message?.content ?? "";
   768→      parsed = extractJSON(raw);
   769→      if (parsed) {
   770→        // emit the repaired sceneText in one chunk so the client catches up
   771→        const { text } = extractPartialSceneText(raw);
   772→        if (text) onDelta(text);
   773→      }
   774→    } catch (e) {
   775→      console.warn("[ai] streaming arcade turn repair attempt failed:", e);
   776→    }
   777→  }
   778→  if (!parsed) {
   779→    console.warn("[ai] streaming arcade turn fallback engaged. head:", brief(raw));
   780→    const fallback: ArcadeTurnResponse = {
   781→      sceneText:
   782→        "The dream stutters for a moment — the scene is still forming. Try describing what you do next.",
   783→      choices: [
   784→        { id: "continue", label: "Press onward into the dream", hint: "Move deeper" },
   785→        { id: "observe", label: "Pause and observe your surroundings" },
   786→      ],
   787→      proposedDelta: { reasoning: "Model response was malformed; no state change applied." },
   788→      discoveredMotifs: [],
   789→    };
   790→    onDelta(fallback.sceneText);
   791→    return { response: fallback, raw };
   792→  }
   793→
   794→  const safe = turnSchema.parse(parsed);
   795→  const response = buildArcadeResponse(safe);
   796→  return { response, raw };
   797→}
