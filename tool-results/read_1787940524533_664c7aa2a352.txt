    13→async function fetchSession(id: string) {
    14→  const res = await fetch(`/api/arcade/sessions/${id}`);
    15→  if (!res.ok) throw new Error("not found");
    16→  return res.json();
    17→}

    22→  "somewhere behind your eyes, the scene is assembling…",
    23→  "a door is deciding whether to open…",
    24→  "the motifs are finding each other again…",
    25→  "the dream is remembering you…",

    29→export function ArcadeSessionView() {
    30→  const sessionId = useApp((s) => s.activeSessionId);
    31→  const navigate = useApp((s) => s.navigate);
    32→  const qc = useQueryClient();

    34→  const [pending, setPending] = useState(false);

    39→  // with text that types itself as Gemini produces it.
    40→  const [streamingScene, setStreamingScene] = useState("");
    41→
    42→  // While a turn is pending: rotate whispers + count seconds, so waiting feels
    43→  // like drifting rather than stalling.
    44→  useEffect(() => {
    45→    if (!pending) return;
    46→    setWhisper(0);
    47→    setElapsedSec(0);
    48→    setStreamingScene("");
    49→    const wi = setInterval(() => setWhisper((w) => (w + 1) % WHISPERS.length), 3600);
    50→    const si = setInterval(() => setElapsedSec((s) => s + 1), 1000);
    51→    return () => {
    52→      clearInterval(wi);
    53→      clearInterval(si);
    54→    };
    55→  }, [pending]);
    56→
    57→  const { data, isLoading } = useQuery({
    58→    queryKey: ["session", sessionId],
    59→    queryFn: () => fetchSession(sessionId!),
    60→    enabled: !!sessionId,
    61→  });
    62→
    63→  if (isLoading || !data) {
    64→    return (
    65→      <div className="flex items-center justify-center py-28">
    66→        <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
    67→      </div>
    68→    );
    69→  }
    70→
    71→  const session = data.session;
    72→  const turns: any[] = session.turns ?? [];
    73→  const lastTurn = turns[turns.length - 1];
    74→  const state: SimulationState = (() => {
    75→    try {
    76→      return JSON.parse(session.stateJson) as SimulationState;
    77→    } catch {
    78→      return { fear: 25, lucidity: 40, stability: 70, agency: 35, turn: turns.length, discoveredMotifs: [], visitedScenes: [], inventory: [], phase: "opening" };
    79→    }
    80→  })();
    81→  const ended = session.status === "ended" && !!session.ending;
    82→  const choices: ArcadeChoice[] = (() => {
    83→    try {
    84→      return JSON.parse(lastTurn?.choicesJson || "[]") as ArcadeChoice[];

    86→      return [];
    87→    }
    88→  })();
    89→  const discovered = (() => {
    90→    try {
    91→      return Array.from(new Set(turns.flatMap((t) => JSON.parse(t.discoveredMotifsJson || "[]")))) as string[];
    92→    } catch {
    93→      return [];
    94→    }
    95→  })();
    96→
    97→  async function takeTurn(payload: { userAction?: string; choiceId?: string }) {
    98→    if (pending || ended) return;
    99→    setPending(true);
   100→    setStreamingScene("");
   101→    try {
   102→      // r6: SSE streaming turn — the scene text types itself as the model
   103→      // produces it. The endpoint emits delta + final events; we parse the
   104→      // text/event-stream manually (no EventSource because we need POST).
   105→      const res = await fetch(`/api/arcade/sessions/${session.id}/turn/stream`, {
   106→        method: "POST",
   107→        headers: { "Content-Type": "application/json" },
   108→        body: JSON.stringify(payload),
   109→      });
   110→      if (!res.ok) {
   111→        const err = await res.json().catch(() => ({}));
   112→        if (res.status === 409) {
   113→          // already ended — refresh
   114→          qc.invalidateQueries({ queryKey: ["session", session.id] });
   115→          return;
   116→        }
   117→        throw new Error(err.error || "The dream faltered.");
   118→      }
   119→      if (!res.body) throw new Error("The dream faltered.");
   120→      const reader = res.body.getReader();
   121→      const decoder = new TextDecoder();
   122→      let buffer = "";
   123→      let finalPayload: any = null;
   124→      let errorMsg: string | null = null;
   125→      while (true) {
   126→        const { value, done } = await reader.read();
   127→        if (done) break;
   128→        buffer += decoder.decode(value, { stream: true });
   129→        let idx;
   130→        while ((idx = buffer.indexOf("\n\n")) !== -1) {
   131→          const chunk = buffer.slice(0, idx);
   132→          buffer = buffer.slice(idx + 2);
   133→          for (const line of chunk.split("\n")) {
   134→            const m = line.match(/^data:\s?(.*)$/);
   135→            if (!m) continue;
   136→            try {
   137→              const ev = JSON.parse(m[1]);
   138→              if (ev.type === "delta" && typeof ev.text === "string") {
   139→                setStreamingScene((s) => s + ev.text);
   140→              } else if (ev.type === "final") {
   141→                finalPayload = ev;
   142→              } else if (ev.type === "error") {
   143→                errorMsg = ev.error || "The dream faltered.";
   144→              }
   145→            } catch {
   146→              // skip malformed
   147→            }
   148→          }
   149→        }
   150→      }
   151→      if (errorMsg) throw new Error(errorMsg);
   152→      // If we got no final event, the stream ended early — refresh from server.
   153→      if (!finalPayload) {
   154→        qc.invalidateQueries({ queryKey: ["session", session.id] });
   155→        qc.invalidateQueries({ queryKey: ["sessions"] });
   156→        setAction("");
   157→        return;
   158→      }
   159→      qc.invalidateQueries({ queryKey: ["session", session.id] });
   160→      qc.invalidateQueries({ queryKey: ["sessions"] });
   161→      setAction("");
   162→      if (finalPayload.ending) {
   163→        toast({ title: finalPayload.ending.title, description: finalPayload.ending.body });
   164→      }
   165→    } catch (e: any) {
   166→      toast({ title: "Turn failed", description: e.message, variant: "destructive" });

   168→      setPending(false);
   169→      setStreamingScene("");
   170→    }
   171→  }
   172→
   173→  return (
   174→    <div className="mx-auto w-full max-w-4xl px-5 sm:px-8 py-8 sm:py-12">
   175→      <button

   191→          <h1 className="font-display tracking-display text-4xl sm:text-5xl leading-tight balance">

   193→          </h1>
   194→        </div>
   195→      </div>
   196→
   197→      {/* state meters — always visible */}

   200→          <Meter label="Fear" value={state.fear} tone="tense" />

   202→          <Meter label="Stability" value={state.stability} tone="neutral" />

   314→                      background:

   452→            <p className="mt-2 text-[11px] text-muted-foreground">

   455→            </p>
   456→          </motion.div>

   463→// r10 — Story share controls for an ENDED session. Creates a read-only public
   464→// link to the session's narrative (#/story/<token>) that works signed-out.
   465→// The share never exposes the dream's raw text — only the story of the
   466→// re-entry: mode, every turn (action + scene), the ending, the final meters.
   467→function StoryShare({ session }: { session: any }) {
   468→  const { toast } = useToast();
   469→  const qc = useQueryClient();
   470→  const navigate = useApp((s) => s.navigate);
   471→  const [busy, setBusy] = useState(false);
   472→  const [copied, setCopied] = useState(false);
   473→
   474→  const token: string | null = session.shareToken ?? null;
   475→  const url =
   476→    token && typeof window !== "undefined" ? `${window.location.origin}/#/story/${token}` : "";
   477→
   478→  async function onShare() {
   479→    if (busy) return;
   480→    setBusy(true);
   481→    try {
   482→      const res = await fetch(`/api/arcade/sessions/${session.id}/share`, { method: "POST" });
   483→      const body = await res.json().catch(() => ({}));
   484→      if (!res.ok) throw new Error(body.error || "The story could not be shared.");
   485→      await qc.invalidateQueries({ queryKey: ["session", session.id] });
   486→      const link = body?.share?.token
   487→        ? `${window.location.origin}/#/story/${body.share.token}`
   488→        : "";
   489→      if (link && navigator.clipboard) {
   490→        navigator.clipboard.writeText(link).catch(() => {});
   491→      }
   492→      toast({
   493→        title: "Story shared",
   494→        description: "A read-only link was copied to your clipboard. The dream's raw memory stays private.",
   495→      });
   496→    } catch (e: any) {
   497→      toast({ title: "Sharing failed", description: e.message, variant: "destructive" });
   498→    } finally {
   499→      setBusy(false);
   500→    }
   501→  }
   502→
   503→  async function onRevoke() {
   504→    if (busy) return;
   505→    setBusy(true);
   506→    try {
   507→      const res = await fetch(`/api/arcade/sessions/${session.id}/share`, { method: "DELETE" });
   508→      if (!res.ok) throw new Error("The link could not be revoked.");
   509→      await qc.invalidateQueries({ queryKey: ["session", session.id] });
   510→      toast({
   511→        title: "Story withdrawn",
   512→        description: "The link no longer resolves. The story returns to private memory.",
   513→      });
   514→    } catch (e: any) {
   515→      toast({ title: "Revoke failed", description: e.message, variant: "destructive" });
   516→    } finally {
   517→      setBusy(false);
   518→    }
   519→  }
   520→
   521→  async function onCopy() {
   522→    if (!url) return;
   523→    try {
   524→      await navigator.clipboard.writeText(url);
   525→      setCopied(true);
   526→      setTimeout(() => setCopied(false), 1600);
   527→    } catch {
   528→      /* clipboard unavailable */
   529→    }
   530→  }
   531→
   532→  if (!token) {
   533→    return (
   534→      <div className="mt-5">
   535→        <Button
   536→          variant="outline"
   537→          size="sm"
   538→          onClick={onShare}
   539→          disabled={busy}
   540→          className="h-9 story-share-btn"
   541→          aria-label="Share this session as a read-only story"
   542→        >
   543→          {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Share2 className="h-4 w-4" strokeWidth={1.6} />}
   544→          <span className="ml-1.5">Share this story</span>
   545→        </Button>
   546→        <p className="mt-2 text-[11px] text-muted-foreground pretty max-w-md mx-auto">
   547→          A read-only page with the whole re-entry — your choices, the scenes, the ending. The
   548→          recorded dream itself is never included.
   549→        </p>
   550→      </div>
   551→    );
   552→  }
   553→
   554→  return (
   555→    <div className="mt-5 story-share-row">
   556→      <div className="flex items-center justify-center gap-2 flex-wrap">
   557→        <button
   558→          onClick={onCopy}
   559→          className="story-link-chip inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs focus-ring"
   560→          aria-label="Copy the story link"
   561→          title={url}
   562→        >
   563→          {copied ? (
   564→            <Check className="h-3 w-3 text-foreground" strokeWidth={2} />
   565→          ) : (
   566→            <Copy className="h-3 w-3" strokeWidth={1.7} />
   567→          )}
   568→          {copied ? "copied" : "story link"}
   569→        </button>
   570→        <button
   571→          onClick={() => navigate("story", { shareToken: token })}
   572→          className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs border border-border hover:bg-foreground/[0.04] transition focus-ring"
   573→        >
   574→          <BookOpenText className="h-3 w-3" strokeWidth={1.7} />
   575→          Read as story
   576→        </button>
   577→        <button
   578→          onClick={onRevoke}
   579→          disabled={busy}
   580→          className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs text-muted-foreground hover:text-destructive transition focus-ring disabled:opacity-50"
   581→          aria-label="Withdraw the story link"
   582→        >
   583→          <Link2Off className="h-3 w-3" strokeWidth={1.7} />
   584→          withdraw
   585→        </button>
   586→      </div>
   587→      <p className="mt-2 text-[11px] text-muted-foreground">
   588→        Shared {session.sharedAt ? new Date(session.sharedAt).toLocaleDateString(undefined, { month: "short", day: "numeric" }) : "recently"} · read-only · the recorded dream stays private
   589→      </p>
   590→    </div>
   591→  );
   592→}

   594→function Meter({ label, value, tone }: { label: string; value: number; tone: "tense" | "lucid" | "neutral" }) {
   595→  const color =
   596→    tone === "tense"
   597→      ? "linear-gradient(90deg, #b1a6a4, #413f3d)"
   598→      : tone === "lucid"
   599→      ? "linear-gradient(90deg, #d8cfd0, #697184)"
   600→      : "linear-gradient(90deg, #d8cfd0, #b1a6a4)";
   601→  return (
   602→    <div>
   603→      <div className="flex items-center justify-between text-xs mb-1.5">
   604→        <span className="text-muted-foreground">{label}</span>
   605→        <span className="font-data text-muted-foreground">{value.toFixed(0)}</span>
   606→      </div>
   607→      <div className="meter-track">
   608→        <div className="meter-fill" style={{ transform: `scaleX(${value / 100})`, background: color }} />
   609→      </div>
   610→    </div>
   611→  );
   612→}
