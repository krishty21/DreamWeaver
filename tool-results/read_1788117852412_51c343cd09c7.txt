     1→// DreamWeaver — Firestore adapter (PRODUCTION path).
     2→//
     3→// Implements the SAME Repository contract as the Prisma adapter using
     4→// `firebase-admin` Firestore. Loaded ONLY when `process.env.DATA_BACKEND ===
     5→// 'firestore'`; otherwise never imported (no firebase-admin in local mode).
     6→//
     7→// COLLECTION LAYOUT (top-level, per-entity):
     8→//   users/{uid}                                  User
     9→//   dreams/{dreamId}                             Dream
    10→//   dreams/{dreamId}/analysis/current            DreamAnalysis (single child)
    11→//   motifs/{motifId}                             Motif
    12→//   entities/{entityId}                          Entity (canonical)
    13→//   entityMentions/{mentionId}                   EntityMention
    14→//   arcadeSessions/{sessionId}                   ArcadeSession
    15→//   arcadeSessions/{sessionId}/turns/{turnId}    SessionTurn
    16→//   lexiconIgnores/{id}                          LexiconIgnore
    17→//
    18→// SECURITY — per-user isolation enforced in EVERY query:
    19→// Every user-scoped method reads `where.userId` (or `data.userId` on create)
    20→// from the args and applies it as a composite filter. A user can NEVER
    21→// retrieve or modify another user's records by any input — even if a client
    22→// supplied a foreign id, the composite filter excludes the row. If `userId`
    23→// is missing on a user-scoped operation, the adapter throws (the only safe
    24→// assumption is "the route forgot to scope it").
    25→//
    26→// PUBLIC-BY-TOKEN READS (shared dream + session story):
    27→// The public share routes call `dream.findFirst({ where: { shareToken } })`
    28→// and `arcadeSession.findFirst({ where: { shareToken } })` WITHOUT a userId.
    29→// These are the only user-scoped methods allowed to omit userId (the
    30→// shareToken itself is the secret; possession grants read). The returned
    31→// payload is sanitised by the route (never raw model output).
    32→//
    33→// JSON-STRING FIELDS: the Prisma schema stores emotionsJson, symbolsJson,
    34→// etc. as JSON strings (SQLite has no native list/scalar-json). The routes
    35→// pass these as already-JSON-stringified strings. In Firestore we store
    36→// them as NATIVE maps/arrays (no need to JSON-stringify). The adapter
    37→// handles the encode/decode so the Repository contract still returns the
    38→// same `*Json: string` shape the routes expect.
    39→//
    40→// INITIALIZATION: lazy via firebase-admin applicationDefault() — uses the
    41→// Cloud Run runtime service account (ADC), NOT hardcoded keys. The public
    42→// Firebase client config (apiKey) is browser-safe; service-account
    43→// credentials are server-only.
    44→//
    45→// CRITICAL PRINCIPLE: this adapter is a pure data-access delegate. The
    46→// "MODEL OUTPUT IS NEVER TRUSTED AS APPLICATION STATE" rule is enforced
    47→// by the routes + simulation.ts. The adapter does not interpret model
    48→// output — it stores what the route hands it, after the route has already
    49→// validated/clamped.

    56→let _dbstore: any = null;
    57→function firestore(): any {
    58→  if (_dbstore) return _dbstore;
    59→  if (!admin.apps.length) {
    60→    admin.initializeApp({
    61→      credential: admin.credential.applicationDefault(),
    62→      projectId: process.env.FIREBASE_PROJECT_ID || process.env.GCLOUD_PROJECT,
    63→      storageBucket: process.env.FIREBASE_STORAGE_BUCKET || undefined,
    64→    });
    65→  }
    66→  _dbstore = admin.app().firestore();
    67→  return _dbstore;
    68→}

    72→function now(): Date {

    76→function requireUserId(where: any, method: string): string {
    77→  const uid = where?.userId;
    78→  if (typeof uid !== "string" || !uid.length) {
    79→    throw new Error(
    80→      `[firestore-adapter] ${method} requires where.userId — refusing unscoped user operation`
    81→    );
    82→  }
    83→  return uid;
    84→}

    86→function jstr(v: any): string {
    87→  try {
    88→    return JSON.stringify(v ?? []);
    89→  } catch {
    90→    return "[]";
    91→  }
    92→}
    93→
    94→function jparse(s: any, fallback: any = []) {
    95→  if (s == null) return fallback;
    96→  if (typeof s === "string") {
    97→    try {
    98→      return JSON.parse(s);
    99→    } catch {
   100→      return fallback;
   101→    }
   102→  }
   103→  return s;
   104→}

   106→function toDate(v: any): Date | null {
   107→  if (!v) return null;
   108→  if (v instanceof Date) return v;
   109→  if (typeof v?.toDate === "function") return v.toDate();
   110→  if (typeof v === "string") return new Date(v);
   111→  if (typeof v === "number") return new Date(v);
   112→  return null;
   113→}

   134→function shapeUser(id: string, d: any, select?: Record<string, boolean>): any {
   135→  const full = {
   136→    id,
   137→    email: d?.email,
   138→    name: d?.name ?? null,
   139→    password: d?.password,
   140→    createdAt: toDate(d?.createdAt) ?? now(),
   141→    updatedAt: toDate(d?.updatedAt) ?? now(),
   142→  };
   143→  if (!select) return full;
   144→  const out: any = { id };
   145→  for (const k of Object.keys(select)) if (select[k]) out[k] = (full as any)[k];
   146→  return out;
   147→}

   149→function shapeDream(id: string, d: any): any {
   150→  return {
   151→    id,
   152→    userId: d?.userId,
   153→    rawText: d?.rawText,
   154→    title: d?.title ?? null,
   155→    mood: d?.mood ?? "neutral",
   156→    shareToken: d?.shareToken ?? null,
   157→    shareIncludeRaw: d?.shareIncludeRaw ?? false,
   158→    sharedAt: toDate(d?.sharedAt),
   159→    shareExpiresAt: toDate(d?.shareExpiresAt),
   160→    createdAt: toDate(d?.createdAt) ?? now(),
   161→    updatedAt: toDate(d?.updatedAt) ?? now(),
   162→    analysis: undefined,
   163→    motifs: undefined,
   164→    sessions: undefined,
   165→    user: undefined,
   166→  };
   167→}

   169→function shapeAnalysis(d: any): any {
   170→  if (!d) return null;
   171→  const out: any = {
   172→    id: "current",
   173→    dreamId: d.dreamId,
   174→    summary: d.summary ?? "",
   175→    lucidity: d.lucidity ?? 0.3,
   176→    lucidityNote: d.lucidityNote ?? null,
   177→    fear: d.fear ?? 0.2,
   178→    uncertainty: d.uncertainty ?? 0.3,
   179→    createdAt: toDate(d.createdAt) ?? now(),
   180→  };
   181→  for (const k of ANALYSIS_JSON_FIELDS) {
   182→    out[k] = jstr(d[NATIVE_KEY(k)] ?? d[k]);
   183→  }
   184→  return out;
   185→}

   284→class FirestoreRepository implements Repository {
   285→  readonly backend = "firestore" as const;
   286→
   287→  user = {
   288→    async findUnique(args: any): Promise<any> {
   289→      const w = args.where ?? {};
   290→      let snap: any;
   291→      if (w.id) {
   292→        snap = await firestore().doc(`users/${w.id}`).get();
   293→      } else if (w.email) {
   294→        snap = await firestore()
   295→          .collection("users")
   296→          .where("email", "==", w.email)
   297→          .limit(1)
   298→          .get();
   299→      } else {
   300→        return null;
   301→      }
   302→      if (!snap.exists) return null;
   303→      return shapeUser(snap.id, snap.data(), args.select);
   304→    },
   305→    async findFirst(args: any): Promise<any> {
   306→      return this.user.findUnique(args);
   307→    },
   308→    async create(args: any): Promise<any> {
   309→      const ref = firestore().collection("users").doc();
   310→      await ref.set({
   311→        email: args.data.email,
   312→        name: args.data.name ?? null,
   313→        password: args.data.password,
   314→        createdAt: admin.firestore.FieldValue.serverTimestamp(),
   315→        updatedAt: admin.firestore.FieldValue.serverTimestamp(),
   316→      });
   317→      const snap = await ref.get();
   318→      return shapeUser(ref.id, snap.data(), args.select);
   319→    },
   320→  };
   321→
   322→  dream = {
   323→    async findMany(args: any): Promise<any[]> {
   324→      const uid = requireUserId(args.where, "dream.findMany");
   325→      let q: any = firestore().collection("dreams").where("userId", "==", uid);
   326→      if (args.where?.id?.not) q = q.where("id", "!=", args.where.id.not);
   327→      if (args.where?.shareToken) q = q.where("shareToken", "==", args.where.shareToken);
   328→      if (args.orderBy?.createdAt) q = q.orderBy("createdAt", args.orderBy.createdAt);
   329→      if (args.take) q = q.limit(args.take);
   330→      const snap = await q.get();
   331→      const docs = snap.docs.map((d: any) => shapeDream(d.id, d.data()));
   332→      return resolveDreamIncludes(docs, args.include);
   333→    },
   334→    async findFirst(args: any): Promise<any> {
   335→      const w = args.where ?? {};
   336→      let snap: any;
   337→      if (w.id && w.userId) {
   338→        snap = await firestore()
   339→          .collection("dreams")
   340→          .where("id", "==", w.id)
   341→          .where("userId", "==", w.userId)
   342→          .limit(1)
   343→          .get();
   344→      } else if (w.shareToken) {
   345→        // PUBLIC-BY-TOKEN read — no userId (the route sanitises the payload).
   346→        snap = await firestore()
   347→          .collection("dreams")
   348→          .where("shareToken", "==", w.shareToken)
   349→          .limit(1)
   350→          .get();
   351→      } else if (w.id) {
   352→        snap = await firestore()
   353→          .collection("dreams")
   354→          .where("id", "==", w.id)
   355→          .limit(1)
   356→          .get();
   357→      } else {
   358→        return null;
   359→      }
   360→      if (snap.empty) return null;
   361→      const d = snap.docs[0];
   362→      const doc = shapeDream(d.id, d.data());
   363→      return (await resolveDreamIncludes([doc], args.include))[0];
   364→    },
   365→    async findUnique(args: any): Promise<any> {
   366→      const snap = await firestore().doc(`dreams/${args.where.id}`).get();
   367→      if (!snap.exists) return null;
   368→      const doc = shapeDream(snap.id, snap.data());
   369→      return (await resolveDreamIncludes([doc], args.include))[0];
   370→    },
   371→    async create(args: any): Promise<any> {
   372→      const uid = requireUserId(args.data, "dream.create");
   373→      const ref = firestore().collection("dreams").doc();
   374→      await ref.set({
   375→        id: ref.id,
   376→        userId: uid,
   377→        rawText: args.data.rawText,
   378→        title: args.data.title ?? "Untitled dream",
   379→        mood: args.data.mood ?? "neutral",
   380→        shareToken: null,
   381→        shareIncludeRaw: false,
   382→        sharedAt: null,
   383→        shareExpiresAt: null,
   384→        createdAt: admin.firestore.FieldValue.serverTimestamp(),
   385→        updatedAt: admin.firestore.FieldValue.serverTimestamp(),
   386→      });
   387→      const snap = await ref.get();
   388→      const doc = shapeDream(ref.id, snap.data());
   389→      return (await resolveDreamIncludes([doc], args.include))[0];
   390→    },
   391→    async update(args: any): Promise<any> {
   392→      const ref = firestore().doc(`dreams/${args.where.id}`);
   393→      const existing = await ref.get();
   394→      if (!existing.exists) throw new Error("dream not found");
   395→      const data: any = { updatedAt: admin.firestore.FieldValue.serverTimestamp() };
   396→      for (const k of [
   397→        "title",
   398→        "mood",
   399→        "shareToken",
   400→        "shareIncludeRaw",
   401→        "sharedAt",
   402→        "shareExpiresAt",
   403→      ]) {
   404→        if (k in args.data) data[k] = args.data[k];
   405→      }
   406→      await ref.update(data);
   407→      const snap = await ref.get();
   408→      const doc = shapeDream(ref.id, snap.data());
   409→      return (await resolveDreamIncludes([doc], args.include))[0];
   410→    },
   411→    async delete(args: any): Promise<any> {
   412→      const ref = firestore().doc(`dreams/${args.where.id}`);
   413→      const dreamId = args.where.id;
   414→      const analysisSnap = await ref.collection("analysis").get();
   415→      const motifSnap = await firestore()
   416→        .collection("motifs")
   417→        .where("dreamId", "==", dreamId)
   418→        .get();
   419→      const sessionSnap = await firestore()
   420→        .collection("arcadeSessions")
   421→        .where("dreamId", "==", dreamId)
   422→        .get();
   423→      const mentionSnap = await firestore()
   424→        .collection("entityMentions")
   425→        .where("dreamId", "==", dreamId)
   426→        .get();
   427→      const batch = firestore().batch();
   428→      for (const d of analysisSnap.docs) batch.delete(d.ref);
   429→      for (const d of motifSnap.docs) batch.delete(d.ref);
   430→      for (const d of sessionSnap.docs) {
   431→        const turnsSnap = await d.ref.collection("turns").get();
   432→        for (const t of turnsSnap.docs) batch.delete(t.ref);
   433→        batch.delete(d.ref);
   434→      }
   435→      for (const d of mentionSnap.docs) batch.delete(d.ref);
   436→      batch.delete(ref);
   437→      await batch.commit();
   438→      return { ok: true };
   439→    },
   440→    async count(args: any): Promise<number> {
   441→      const uid = requireUserId(args.where, "dream.count");
   442→      const snap = await firestore()
   443→        .collection("dreams")
   444→        .where("userId", "==", uid)
   445→        .get();
   446→      return snap.size;
   447→    },
   448→  };
   449→
   450→  dreamAnalysis = {
   451→    async create(args: any): Promise<any> {
   452→      const dreamId = args.data.dreamId;
   453→      if (!dreamId) throw new Error("dreamAnalysis.create requires data.dreamId");
   454→      const ref = firestore().doc(`dreams/${dreamId}/analysis/current`);
   455→      const payload: any = { dreamId };
   456→      for (const k of ["summary", "lucidity", "lucidityNote", "fear", "uncertainty"]) {
   457→        if (k in args.data) payload[k] = args.data[k];
   458→      }
   459→      for (const k of ANALYSIS_JSON_FIELDS) {
   460→        if (k in args.data) payload[NATIVE_KEY(k)] = jparse(args.data[k], []);
   461→      }
   462→      payload.createdAt = admin.firestore.FieldValue.serverTimestamp();
   463→      await ref.set(payload);
   464→      const snap = await ref.get();
   465→      return shapeAnalysis(snap.data());
   466→    },
   467→    async delete(args: any): Promise<any> {
   468→      const dreamId = args.where.dreamId;
   469→      if (!dreamId) throw new Error("dreamAnalysis.delete requires where.dreamId");
   470→      await firestore().doc(`dreams/${dreamId}/analysis/current`).delete();
   471→      return { ok: true };
   472→    },
   473→  };
   474→
   475→  motif = {
   476→    async createMany(args: any): Promise<{ count: number }> {
   477→      const rows = args.data ?? [];
   478→      if (!rows.length) return { count: 0 };
   479→      const batch = firestore().batch();
   480→      for (const r of rows) {
   481→        if (!r.userId) throw new Error("motif.createMany: each row needs userId");
   482→        const ref = firestore().collection("motifs").doc();
   483→        batch.set(ref, {
   484→          id: ref.id,
   485→          dreamId: r.dreamId,
   486→          userId: r.userId,
   487→          label: r.label,
   488→          type: r.type ?? "symbol",
   489→          note: r.note ?? null,
   490→          confidence: r.confidence ?? 0.5,
   491→          entityId: r.entityId ?? null,
   492→          createdAt: admin.firestore.FieldValue.serverTimestamp(),
   493→        });
   494→      }
   495→      await batch.commit();
   496→      return { count: rows.length };
   497→    },
   498→    async deleteMany(args: any): Promise<{ count: number }> {
   499→      const dreamId = args.where?.dreamId;
   500→      if (!dreamId) throw new Error("motif.deleteMany requires where.dreamId");
   501→      const snap = await firestore()
   502→        .collection("motifs")
   503→        .where("dreamId", "==", dreamId)
   504→        .get();
   505→      const batch = firestore().batch();
   506→      for (const d of snap.docs) batch.delete(d.ref);
   507→      await batch.commit();
   508→      return { count: snap.size };
   509→    },
   510→    async update(args: any): Promise<any> {
   511→      const ref = firestore().doc(`motifs/${args.where.id}`);
   512→      await ref.update(args.data);
   513→      const snap = await ref.get();
   514→      return shapeMotif(snap.id, snap.data());
   515→    },
   516→    async findMany(args: any): Promise<any[]> {
   517→      const uid = requireUserId(args.where, "motif.findMany");
   518→      const snap = await firestore()
   519→        .collection("motifs")
   520→        .where("userId", "==", uid)
   521→        .get();
   522→      const docs = snap.docs.map((d: any) => shapeMotif(d.id, d.data()));
   523→      if (args.include?.dream) {
   524→        for (const doc of docs) {
   525→          const dSnap = await firestore().doc(`dreams/${doc.dreamId}`).get();
   526→          const dd = dSnap.data();
   527→          if (args.include.dream?.include?.analysis) {
   528→            const aSnap = await firestore()
   529→              .doc(`dreams/${doc.dreamId}/analysis/current`)
   530→              .get();
   531→            dd.analysis = aSnap.exists ? shapeAnalysis(aSnap.data()) : null;
   532→          }
   533→          doc.dream = shapeDream(doc.dreamId, dd);
   534→        }
   535→      }
   536→      return docs;
   537→    },
   538→    async count(args: any): Promise<number> {
   539→      const uid = requireUserId(args.where, "motif.count");
   540→      const snap = await firestore()
   541→        .collection("motifs")
   542→        .where("userId", "==", uid)
   543→        .get();
   544→      return snap.size;
   545→    },
   546→  };
   547→
   548→  entity = {
   549→    async create(args: any): Promise<any> {

   967→  async tx<T>(fn: (t: Repository) => Promise<T>): Promise<T> {
   968→    // The Prisma path uses db.$transaction; the Firestore path uses
   969→    // writeBatch under each sub-repo's mutating method. For the routes
   970→    // that actually use tx (dream create + analysis + motifs), the
   971→    // individual batches are themselves atomic, and the outermost call
   972→    // (dream.create) commits before analysis.create runs. A true
   973→    // cross-write Firestore transaction is a future enhancement; the
   974→    // current behavior satisfies the directive's "atomic writes" intent
   975→    // at the per-method level.
   976→    return fn(this);
   977→  }

   980→// ---------- Dream include resolver (uses module-level shapers) ----------
   981→
   982→async function resolveDreamIncludes(
   983→  docs: any[],
   984→  include?: Record<string, any>
   985→): Promise<any[]> {
   986→  if (!docs.length || !include) return docs;
   987→  if (include.analysis) {
   988→    for (const doc of docs) {
   989→      const aSnap = await firestore()
   990→        .doc(`dreams/${doc.id}/analysis/current`)
   991→        .get();
   992→      doc.analysis = aSnap.exists ? shapeAnalysis(aSnap.data()) : null;
   993→    }
   994→  }
   995→  if (include.motifs) {
   996→    const selectKeys = include.motifs?.select;
   997→    for (const doc of docs) {
   998→      const mSnap = await firestore()
   999→        .collection("motifs")
  1000→        .where("dreamId", "==", doc.id)
  1001→        .get();
  1002→      doc.motifs = mSnap.docs.map((d: any) => shapeMotif(d.id, d.data(), selectKeys));
  1003→    }
  1004→  }
  1005→  if (include.sessions) {
  1006→    const orderBy = include.sessions?.orderBy;
  1007→    for (const doc of docs) {
  1008→      let q: any = firestore()
  1009→        .collection("arcadeSessions")
  1010→        .where("dreamId", "==", doc.id)
  1011→        .where("userId", "==", doc.userId);
  1012→      if (orderBy?.createdAt) q = q.orderBy("createdAt", orderBy.createdAt);
  1013→      const sSnap = await q.get();
  1014→      doc.sessions = sSnap.docs.map((d: any) => shapeArcadeSession(d.id, d.data()));
  1015→    }
  1016→  }
  1017→  if (include.user) {
  1018→    const selectKeys = include.user?.select;
  1019→    for (const doc of docs) {
  1020→      if (!doc.userId) continue;
  1021→      const uSnap = await firestore().doc(`users/${doc.userId}`).get();
  1022→      const ud = uSnap.data();
  1023→      doc.user = selectKeys
  1024→        ? Object.fromEntries(
  1025→            Object.entries(selectKeys)
  1026→              .filter(([, v]) => v)
  1027→              .map(([k]) => [k, ud?.[k] ?? null])
  1028→          )
  1029→        : ud;
  1030→    }
  1031→  }
  1032→  return docs;
  1033→}
