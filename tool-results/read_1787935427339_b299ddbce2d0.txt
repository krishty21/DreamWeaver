    43→async function fetchDream(id: string) {
    44→  const res = await fetch(`/api/dreams/${id}`);
    45→  if (!res.ok) throw new Error("not found");
    46→  return res.json();
    47→}

    49→export function DreamDetailView() {
    50→  const dreamId = useApp((s) => s.activeDreamId);
    51→  const navigate = useApp((s) => s.navigate);
    52→  const { toast } = useToast();
    53→  const qc = useQueryClient();
    54→  const [reflecting, setReflecting] = useState(false);
    55→  const [shareBusy, setShareBusy] = useState(false);
    56→
    57→  const { data, isLoading } = useQuery({
    58→    queryKey: ["dream", dreamId],
    59→    queryFn: () => fetchDream(dreamId!),
    60→    enabled: !!dreamId,
    61→  });
    62→
    63→  if (isLoading || !data) {
    64→    return (
    65→      <div className="flex items-center justify-center py-28">
    66→        <Loader2 className="h-5 w-5 animate-spin text-muted-foreground" />
    67→      </div>
    68→    );
    69→  }
    70→
    71→  const dream = data.dream;
    72→  const a = dream.analysis ? (parseAnalysis(dream.analysis) as DreamAnalysisData) : null;
    73→
    74→  // ---------- share management ----------
    75→  // The share is a read-only public link to this dream's SANITISED reflection.
    76→  // rawText is only exposed if the dreamer explicitly opts in.
    77→  const shareToken: string | null = dream.shareToken ?? null;
    78→  const shareUrl =
    79→    shareToken && typeof window !== "undefined"
    80→      ? `${window.location.origin}/#/shared/${shareToken}`
    81→      : "";
    82→
    83→  async function onShare(opts?: { includeRaw?: boolean; expiresInDays?: number | null }) {
    84→    if (!dream || shareBusy) return;
    85→    setShareBusy(true);
    86→    try {
    87→      const res = await fetch(`/api/dreams/${dream.id}/share`, {
    88→        method: "POST",
    89→        headers: { "Content-Type": "application/json" },
    90→        body: JSON.stringify(opts ?? {}),
    91→      });
    92→      const body = await res.json().catch(() => ({}));
    93→      if (!res.ok) throw new Error(body.error || "The share link could not be created.");
    94→      await qc.invalidateQueries({ queryKey: ["dream", dream.id] });
    95→      if (opts?.expiresInDays !== undefined) {
    96→        toast({
    97→          title: "Link window updated",
    98→          description:
    99→            opts.expiresInDays === null
   100→              ? "The link now stays open until you revoke it."
   101→              : `The link will close in ${opts.expiresInDays} day${opts.expiresInDays === 1 ? "" : "s"}.`,
   102→        });
   103→      } else if (opts?.includeRaw === undefined) {
   104→        toast({
   105→          title: "Reflection shared",
   106→          description: "A read-only link has been created. Your raw memory stays private.",
   107→        });
   108→      }
   109→    } catch (e: any) {
   110→      toast({ title: "Sharing failed", description: e.message, variant: "destructive" });
   111→    } finally {
   112→      setShareBusy(false);
   113→    }
   114→  }
   115→
   116→  async function onRevokeShare() {
   117→    if (!dream || shareBusy) return;
   118→    setShareBusy(true);
   119→    try {
   120→      const res = await fetch(`/api/dreams/${dream.id}/share`, { method: "DELETE" });
   121→      if (!res.ok) throw new Error("The link could not be revoked.");
   122→      await qc.invalidateQueries({ queryKey: ["dream", dream.id] });
   123→      toast({
   124→        title: "Link revoked",
   125→        description: "The shared reflection is no longer reachable.",
   126→      });
   127→    } catch (e: any) {
   128→      toast({ title: "Revoke failed", description: e.message, variant: "destructive" });
   129→    } finally {
   130→      setShareBusy(false);
   131→    }
   132→  }
   133→
   134→  // Re-run the Gemini reflection. The raw dream is never modified — only the
   135→  // derived analysis is replaced.
   136→  async function onReflect() {
   137→    if (!dream || reflecting) return;
   138→    setReflecting(true);
   139→    try {
   140→      const res = await fetch(`/api/dreams/${dream.id}/reanalyze`, { method: "POST" });
   141→      if (!res.ok) {
   142→        const err = await res.json().catch(() => ({}));
   143→        throw new Error(err.error || "The reflection could not be produced.");
   144→      }
   145→      qc.invalidateQueries({ queryKey: ["dream", dream.id] });
   146→      qc.invalidateQueries({ queryKey: ["dreams"] });
   147→      qc.invalidateQueries({ queryKey: ["patterns"] });
   148→      toast({
   149→        title: "Dream re-read",
   150→        description: "A fresh reflection has been woven. Your original memory is unchanged.",
   151→      });
   152→    } catch (e: any) {
   153→      toast({ title: "Re-reflection failed", description: e.message, variant: "destructive" });
   154→    } finally {
   155→      setReflecting(false);
   156→    }
   157→  }
   158→
   159→  async function onDelete() {
   160→    if (!dream) return;
   161→    const ok = window.confirm(
   162→      "Delete this dream and all of its analysis, motifs, and arcade sessions? This cannot be undone."
   163→    );
   164→    if (!ok) return;
   165→    const res = await fetch(`/api/dreams/${dream.id}`, { method: "DELETE" });
   166→    if (res.ok) {
   167→      qc.invalidateQueries({ queryKey: ["dreams"] });
   168→      qc.invalidateQueries({ queryKey: ["patterns"] });
   169→      qc.invalidateQueries({ queryKey: ["sessions"] });
   170→      qc.invalidateQueries({ queryKey: ["me"] });
   171→      toast({ title: "Dream deleted", description: "The record has been removed." });
   172→      navigate("journal");
   173→    }
   174→  }
   175→
   176→  return (
   177→    <div className="mx-auto w-full max-w-4xl px-5 sm:px-8 py-10 sm:py-14">
   178→      <div className="flex flex-wrap items-center justify-between gap-x-3 gap-y-2 mb-8">
   179→        <button
   180→          onClick={() => navigate("journal")}
   181→          className="inline-flex items-center gap-2 text-sm text-muted-foreground hover:text-foreground transition focus-ring"
   182→        >
   183→          <ArrowLeft className="h-4 w-4" strokeWidth={1.6} />
   184→          Journal
   185→        </button>
   186→        <div className="flex items-center gap-1.5 sm:gap-2 ml-auto">
   187→          <Button
   188→            onClick={() => navigate("arcade", { dreamId: dream.id })}
   189→            className="h-9 bg-foreground text-background hover:opacity-90"
   190→          >
   191→            <Compass className="h-4 w-4" strokeWidth={1.6} />
   192→            Re-enter dream
   193→          </Button>
   194→          <Button
   195→            variant="outline"
   196→            size="sm"
   197→            onClick={onReflect}
   198→            disabled={reflecting}
   199→            className="h-9"
   200→          >
   201→            {reflecting ? (
   202→              <Loader2 className="h-4 w-4 animate-spin" />
   203→            ) : (
   204→              <RefreshCw className="h-4 w-4" strokeWidth={1.6} />
   205→            )}
   206→            <span className="sr-only sm:not-sr-only sm:ml-1.5">
   207→              {reflecting ? "Reading…" : a ? "Re-reflect" : "Add reflection"}
   208→            </span>
   209→          </Button>
   210→          {a && (
   211→            <Button
   212→              variant="outline"
   213→              size="sm"
   214→              onClick={() => onShare()}
   215→              disabled={shareBusy || reflecting}
   216→              className="h-9"
   217→              aria-label="Share this reflection"
   218→            >
   219→              {shareBusy ? (
   220→                <Loader2 className="h-4 w-4 animate-spin" />
   221→              ) : (
   222→                <Share2 className="h-4 w-4" strokeWidth={1.6} />
   223→              )}
   224→              <span className="sr-only sm:not-sr-only sm:ml-1.5">
   225→                {shareToken ? "Shared…" : "Share"}
   226→              </span>
   227→            </Button>
   228→          )}
   229→          <Button
   230→            variant="outline"
   231→            size="sm"
   232→            onClick={() => {
   233→              const md = buildDreamMarkdown(dream);
   234→              downloadMarkdown(md, `${slugify(dream.title || "dream")}.md`);
   235→              toast({ title: "Dream exported", description: "Saved as a Markdown file." });
   236→            }}
   237→            className="h-9"
   238→            aria-label="Export this dream as a Markdown file"
   239→          >
   240→            <FileDown className="h-4 w-4" strokeWidth={1.6} />
   241→          </Button>
   242→          <Button
   243→            variant="ghost"
   244→            size="sm"
   245→            onClick={onDelete}
   246→            className="text-muted-foreground hover:text-destructive"
   247→          >
   248→            <Trash2 className="h-4 w-4" strokeWidth={1.6} />
   249→            <span className="sr-only sm:not-sr-only sm:ml-1.5">Delete</span>
   250→          </Button>
   251→        </div>
   252→      </div>
   253→
   254→      {/* Share panel — visible while the reflection is shared */}
   255→      {shareToken && (
   256→        <SharePanel
   257→          url={shareUrl}
   258→          includeRaw={!!dream.shareIncludeRaw}
   259→          sharedAt={dream.sharedAt}
   260→          expiresAt={dream.shareExpiresAt ?? null}
   261→          busy={shareBusy}
   262→          onToggleRaw={(v) => onShare({ includeRaw: v })}
   263→          onSetExpiry={(days) => onShare({ expiresInDays: days })}
   264→          onRevoke={onRevokeShare}
   265→          onPreview={() => navigate("shared", { shareToken })}
   266→        />
   267→      )}
   268→
   269→      <motion.div
   270→        initial={{ opacity: 0, y: 12 }}
   271→        animate={{ opacity: 1, y: 0 }}
   272→        transition={{ duration: 0.5 }}
   273→      >
   274→        <div className="text-xs tracking-caps uppercase text-muted-foreground mb-3 flex items-center gap-2">
   275→          <span>{new Date(dream.createdAt).toLocaleDateString(undefined, { weekday: "long", year: "numeric", month: "long", day: "numeric" })}</span>
   276→          {dream.mood && dream.mood !== "neutral" && (
   277→            <span className="chip">{dream.mood}</span>
   278→          )}
   279→        </div>
   280→        <h1 className="font-display tracking-display text-4xl sm:text-5xl leading-tight balance">
   281→          {dream.title || "A dream, partially recalled"}
   282→        </h1>
   283→      </motion.div>
   284→
   285→      {/* OBSERVED — raw memory */}
   286→      <section className="mt-10">
   287→        <SectionLabel icon={Eye} tag="01 · Observed" label="Your raw dream memory" />
   288→        <div className="mt-4 surface p-6 sm:p-7">
   289→          <Quote className="h-5 w-5 text-muted-foreground mb-3" strokeWidth={1.4} />
   290→          <p className="prose-dream whitespace-pre-wrap pretty">{dream.rawText}</p>
   291→        </div>
   292→      </section>
   293→
   294→      {!a ? (
   295→        <section className="mt-12 surface p-6 text-sm text-muted-foreground">
   296→          No structured reflection was produced for this dream. You can still re-enter it as an
   297→          arcade session.
   298→        </section>
   299→      ) : (
   300→        <>
   301→          {/* Summary */}
   302→          <section className="mt-12">
   303→            <SectionLabel icon={Brain} tag="02 · Reflection" label="Summary" />
   304→            <p className="mt-4 font-display text-2xl leading-snug text-foreground pretty">
   305→              {a.summary}
   306→            </p>
   307→          </section>
   308→
   309→          {/* Emotional signature */}
   310→          <section className="mt-12">
   311→            <SectionLabel icon={Sparkles} tag="03 · Emotional signature" label="What you felt" />
   312→            <div className="mt-4 grid grid-cols-1 sm:grid-cols-2 gap-3">
   313→              {a.emotions.length === 0 && <Empty />}
   314→              {a.emotions.map((e, i) => (
   315→                <EmotionBar key={i} emotion={e} />
   316→              ))}
   317→            </div>
   318→          </section>
   319→
   320→          {/* Grid: motifs / symbols / people / places / actions */}
   321→          <div className="mt-12 grid grid-cols-1 md:grid-cols-2 gap-x-8 gap-y-10">
   322→            <ItemGroup title="Symbolic motifs" items={a.motifs} type="symbol" />
   323→            <ItemGroup title="Symbols" items={a.symbols} type="symbol" />

   325→            <ItemGroup title="Locations" items={a.locations} type="place" />
   326→            <ItemGroup title="Actions" items={a.actions} type="action" />
   327→          </div>
   328→
   329→          {/* Relationships within the dream */}
   330→          {a.relationships.length > 0 && (
   331→            <section className="mt-12">
   332→              <SectionLabel icon={Link2} tag="04 · Within this dream" label="Relationships" />
   333→              <div className="mt-4 flex flex-wrap gap-2.5">
   334→                {a.relationships.map((r, i) => (
   335→                  <span
   336→                    key={i}
   337→                    className="surface-quiet px-3.5 py-2 inline-flex flex-wrap items-center gap-x-2 gap-y-0.5 text-sm max-w-full"
   338→                  >
   339→                    <span className="text-foreground min-w-0">{r.from}</span>
   340→                    <span className="text-[10px] tracking-caps uppercase text-muted-foreground border-b border-border pb-0.5">
   341→                      {r.relation}
   342→                    </span>
   343→                    <span className="text-foreground min-w-0">{r.to}</span>
   344→                  </span>
   345→                ))}
   346→              </div>

   349→
   350→          {/* Lucidity / fear / uncertainty meters */}
   351→          <section className="mt-12">
   352→            <SectionLabel icon={Brain} tag="05 · Dream lucidity & emotional tone" label="Estimate" />

   354→              <Meter label="Lucidity" value={a.lucidity * 100} hint={a.lucidityNote} tone="lucid" />

   363→          {/* Historical connections */}

   368→                {a.historicalConnections.map((c, i) => (
   369→                  <div key={i} className="surface-quiet p-4 flex items-start justify-between gap-3">

   371→                      <div className="font-display text-xl">{cap(c.motif)}</div>

   375→                      <span className="font-data">{c.dreamIds.length}</span>{" "}
   376→                      prior dream{c.dreamIds.length === 1 ? "" : "s"}
   377→                    </div>
   378→                  </div>
   379→                ))}
   380→              </div>
   381→            </section>
   382→          )}
   383→
   384→          {/* Interpretations */}
   385→          {a.interpretations.length > 0 && (
   386→            <section className="mt-12">
   387→              <SectionLabel icon={Sparkles} tag="07 · Possible interpretation" label="AI-generated reflection" />
   388→              <div className="mt-4 space-y-3">

   392→              </div>

   425→function SharePanel({
   426→  url,
   427→  includeRaw,
   428→  sharedAt,
   429→  expiresAt,
   430→  busy,
   431→  onToggleRaw,
   432→  onSetExpiry,
   433→  onRevoke,
   434→  onPreview,
   435→}: {
   436→  url: string;
   437→  includeRaw: boolean;
   438→  sharedAt: string | null;
   439→  expiresAt: string | null;
   440→  busy: boolean;
   441→  onToggleRaw: (v: boolean) => void;
   442→  onSetExpiry: (days: number | null) => void;
   443→  onRevoke: () => void;
   444→  onPreview: () => void;
   445→}) {
   446→  const { toast } = useToast();
   447→  const [copied, setCopied] = useState(false);
   448→
   449→  async function copy() {
   450→    try {
   451→      await navigator.clipboard.writeText(url);
   452→      setCopied(true);
   453→      setTimeout(() => setCopied(false), 1800);
   454→    } catch {
   455→      toast({
   456→        title: "Copy failed",
   457→        description: "Select the link text and copy it manually.",
   458→        variant: "destructive",
   459→      });
   460→    }
   461→  }
   462→
   463→  const shared = sharedAt ? new Date(sharedAt).toLocaleDateString() : null;
   464→
   465→  const expiryDate = expiresAt ? new Date(expiresAt) : null;
   466→  const isExpired = !!expiryDate && expiryDate.getTime() < Date.now();
   467→  const daysLeft = expiryDate
   468→    ? Math.ceil((expiryDate.getTime() - Date.now()) / (24 * 60 * 60 * 1000))
   469→    : null;
   470→  const expiryLabel = !expiryDate
   471→    ? "never"
   472→    : isExpired
   473→    ? "expired"
   474→    : daysLeft !== null && daysLeft <= 1
   475→    ? "last day"
   476→    : `${daysLeft} day${daysLeft === 1 ? "" : "s"} left`;
   477→  const currentWindow: "never" | "7" | "30" =
   478→    !expiryDate ? "never" : isExpired ? "7" : (daysLeft ?? 0) > 22 ? "30" : "7";
   479→
   480→  return (
   481→    <motion.div
   482→      initial={{ opacity: 0, y: -6 }}
   483→      animate={{ opacity: 1, y: 0 }}
   484→      transition={{ duration: 0.35 }}
   485→      className="surface p-5 sm:p-6 mb-2"
   486→      role="region"
   487→      aria-label="Public share link for this reflection"
   488→    >
   489→      <div className="flex flex-col sm:flex-row sm:items-center gap-4">
   490→        <div className="flex-1 min-w-0">
   491→          <div className="flex items-center gap-2 text-xs tracking-caps uppercase text-muted-foreground">
   492→            <MoonStar className="h-3.5 w-3.5" strokeWidth={1.6} />
   493→            Public read-only link
   494→            {shared && <span className="normal-case tracking-normal">· since {shared}</span>}
   495→            <span
   496→              className={`normal-case tracking-normal inline-flex items-center gap-1 ${
   497→                isExpired ? "text-destructive" : ""
   498→              }`}
   499→            >
   500→              ·{" "}
   501→              <Hourglass className="h-3 w-3" strokeWidth={1.7} aria-hidden="true" />
   502→              {isExpired ? "expired — renew or revoke" : expiryLabel}
   503→            </span>
   504→          </div>
   505→          <div className={`mt-2 flex items-center gap-2 ${isExpired ? "opacity-60" : ""}`}>
   506→            <code className="share-url text-xs sm:text-sm px-3 py-2 bg-background/70 border border-border rounded-md truncate flex-1 min-w-0" title={url}>
   507→              {url || "…"}
   508→            </code>
   509→            <Button size="sm" variant="outline" className="h-9 shrink-0" onClick={copy} aria-label="Copy share link">
   510→              {copied ? <Check className="h-4 w-4 text-green-700" strokeWidth={1.8} /> : <Copy className="h-4 w-4" strokeWidth={1.6} />}
   511→              <span className="sr-only sm:not-sr-only sm:ml-1.5">{copied ? "Copied" : "Copy"}</span>
   512→            </Button>
   513→          </div>
   514→          {isExpired && (
   515→            <p className="mt-1.5 text-[11px] text-destructive/90 pretty">
   516→              This window has passed — visitors see a closed page. Choose a new window below to re-open it.
   517→            </p>
   518→          )}
   519→        </div>
   520→        <div className="flex items-center gap-2 shrink-0">
   521→          <Button size="sm" variant="outline" className="h-9" onClick={onPreview}>
   522→            <Eye className="h-4 w-4" strokeWidth={1.6} />
   523→            <span className="sr-only sm:not-sr-only sm:ml-1.5">Preview</span>
   524→          </Button>
   525→          <Button
   526→            size="sm"
   527→            variant="ghost"

   743→function parseAnalysis(a: any): DreamAnalysisData {
   744→  const j = (k: string, fallback: any = []) => {
   745→    try {
   746→      const v = JSON.parse(a[k] ?? "[]");
   747→      return Array.isArray(v) ? v : fallback;
   748→    } catch {
   749→      return fallback;
   750→    }
   751→  };
   752→  return {
   753→    title: a.summary ? "" : "",
   754→    summary: a.summary ?? "",
   755→    emotions: j("emotionsJson"),
   756→    symbols: j("symbolsJson"),
   757→    motifs: j("motifsJson"),
   758→    people: j("peopleJson"),
   759→    locations: j("locationsJson"),
   760→    actions: j("actionsJson"),
   761→    lucidity: a.lucidity ?? 0.3,
   762→    lucidityNote: a.lucidityNote ?? undefined,
   763→    fear: a.fear ?? 0.2,
   764→    uncertainty: a.uncertainty ?? 0.3,
   765→    interpretations: j("interpretationsJson"),
   766→    relationships: j("relationshipsJson"),
   767→    historicalConnections: j("historicalConnectionsJson"),
   768→    mood: (a as any).mood ?? "neutral",
   769→  };
   770→}
